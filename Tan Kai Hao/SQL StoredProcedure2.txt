CL SCR;

SET DEFINE OFF;

DECLARE
    v_max_id NUMBER;
BEGIN
    SELECT NVL(MAX(TO_NUMBER(SUBSTR(PackageID, 3))), 0) INTO v_max_id FROM PACKAGE;
    
    BEGIN
        EXECUTE IMMEDIATE 'DROP SEQUENCE seq_package_id';
    EXCEPTION
        WHEN OTHERS THEN
            IF SQLCODE != -2289 THEN 
                RAISE;
            END IF;
    END;
    
    EXECUTE IMMEDIATE 'CREATE SEQUENCE seq_package_id START WITH ' || (v_max_id + 1) || ' INCREMENT BY 1 NOCACHE';
END;
/

SET SERVEROUTPUT ON FORMAT WRAPPED;

CREATE OR REPLACE PROCEDURE prc_create_package (
    p_item1_id IN CHAR, p_qty1 IN NUMBER,
    p_item2_id IN CHAR, p_qty2 IN NUMBER,
    p_discount IN NUMBER,
    p_item3_id IN CHAR DEFAULT NULL, p_qty3 IN NUMBER DEFAULT NULL
)
AS
    v_item1_name ITEM.ItemName%TYPE;
    v_item1_price ITEM.Price%TYPE;
    v_item1_rest ITEM.RestaurantID%TYPE;
    
    v_item2_name ITEM.ItemName%TYPE;
    v_item2_price ITEM.Price%TYPE;
    v_item2_rest ITEM.RestaurantID%TYPE;

    v_item3_name ITEM.ItemName%TYPE;
    v_item3_price ITEM.Price%TYPE;
    v_item3_rest ITEM.RestaurantID%TYPE;

    v_rest_name RESTAURANT.RestaurantName%TYPE;

    v_pkg_id PACKAGE.PackageID%TYPE;
    v_pkg_name PACKAGE.PackageName%TYPE;
    
    v_orig_price NUMBER(8,2) := 0;
    v_disc_price NUMBER(8,2) := 0;

BEGIN
    DBMS_OUTPUT.PUT_LINE('====================================================================================================');
    DBMS_OUTPUT.PUT_LINE('PARSED PACKAGE PARAMETERS');
    DBMS_OUTPUT.PUT_LINE('>> Item 1   : ' || p_item1_id || ' (Qty: ' || p_qty1 || ')');
    DBMS_OUTPUT.PUT_LINE('>> Item 2   : ' || p_item2_id || ' (Qty: ' || p_qty2 || ')');
    IF p_item3_id IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('>> Item 3   : ' || p_item3_id || ' (Qty: ' || p_qty3 || ')');
    END IF;
    DBMS_OUTPUT.PUT_LINE('>> Discount : ' || (p_discount * 100) || '%');

    IF p_discount <= 0 OR p_discount > 1 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Validation Failed: Discount format invalid. Use decimal between 0.01 and 1.00');
    END IF;

    IF NVL(p_qty1, 0) <= 0 THEN
        RAISE_APPLICATION_ERROR(-20002, 'Validation Failed: Quantity for Item 1 must be greater than 0.');
    END IF;

    IF NVL(p_qty2, 0) <= 0 THEN
        RAISE_APPLICATION_ERROR(-20002, 'Validation Failed: Quantity for Item 2 must be greater than 0.');
    END IF;

    IF p_item3_id IS NOT NULL AND NVL(p_qty3, 0) <= 0 THEN
        RAISE_APPLICATION_ERROR(-20003, 'Validation Failed: Quantity for Item 3 must be greater than 0 if provided.');
    END IF;

    IF p_item1_id = p_item2_id OR (p_item3_id IS NOT NULL AND (p_item3_id = p_item1_id OR p_item3_id = p_item2_id)) THEN
        RAISE_APPLICATION_ERROR(-20004, 'Validation Failed: Duplicate Item IDs detected in parameters.');
    END IF;

    BEGIN
        SELECT ItemName, Price, RestaurantID INTO v_item1_name, v_item1_price, v_item1_rest
        FROM ITEM WHERE ItemID = p_item1_id AND IsAvailable = 'Y';
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20005, 'Validation Failed: Item 1 (' || p_item1_id || ') does not exist or is unavailable.');
    END;

    BEGIN
        SELECT ItemName, Price, RestaurantID INTO v_item2_name, v_item2_price, v_item2_rest
        FROM ITEM WHERE ItemID = p_item2_id AND IsAvailable = 'Y';
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20006, 'Validation Failed: Item 2 (' || p_item2_id || ') does not exist or is unavailable.');
    END;

    IF v_item1_rest != v_item2_rest THEN
        RAISE_APPLICATION_ERROR(-20007, 'Validation Failed: Item 1 and Item 2 belong to completely different restaurants.');
    END IF;

    IF p_item3_id IS NOT NULL THEN
        BEGIN
            SELECT ItemName, Price, RestaurantID INTO v_item3_name, v_item3_price, v_item3_rest
            FROM ITEM WHERE ItemID = p_item3_id AND IsAvailable = 'Y';
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(-20008, 'Validation Failed: Item 3 (' || p_item3_id || ') does not exist or is unavailable.');
        END;

        IF v_item3_rest != v_item1_rest THEN
            RAISE_APPLICATION_ERROR(-20009, 'Validation Failed: Item 3 belongs to a different restaurant from Item 1 & 2.');
        END IF;
    END IF;

    BEGIN
        SELECT RestaurantName INTO v_rest_name FROM RESTAURANT WHERE RestaurantID = v_item1_rest;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            v_rest_name := 'Unknown Restaurant';
    END;

    SELECT 'PK' || LPAD(seq_package_id.NEXTVAL, 5, '0') INTO v_pkg_id FROM DUAL;

    v_orig_price := (v_item1_price * p_qty1) + (v_item2_price * p_qty2);
    v_pkg_name := v_item1_name || ' & ' || v_item2_name;

    IF p_item3_id IS NOT NULL THEN
        v_orig_price := v_orig_price + (v_item3_price * p_qty3);
        v_pkg_name := v_pkg_name || ' & ' || v_item3_name;
    END IF;

    v_pkg_name := v_pkg_name || ' Combo';
    
    v_disc_price := v_orig_price * (1 - p_discount);

    INSERT INTO PACKAGE (PackageID, PackageName, OriginalPrice, Discount)
    VALUES (v_pkg_id, v_pkg_name, v_orig_price, p_discount);
    
    INSERT INTO PACKAGEITEM (PackageID, ItemID, Quantity) VALUES (v_pkg_id, p_item1_id, p_qty1);
    INSERT INTO PACKAGEITEM (PackageID, ItemID, Quantity) VALUES (v_pkg_id, p_item2_id, p_qty2);
    
    IF p_item3_id IS NOT NULL THEN
        INSERT INTO PACKAGEITEM (PackageID, ItemID, Quantity) VALUES (v_pkg_id, p_item3_id, p_qty3);
    END IF;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(' PACKAGE PROFILE ');
    DBMS_OUTPUT.PUT_LINE('  * Package ID   : ' || v_pkg_id);
    DBMS_OUTPUT.PUT_LINE('  * Package Name : ' || v_pkg_name);
    DBMS_OUTPUT.PUT_LINE('  * Restaurant   : ' || v_rest_name);
    DBMS_OUTPUT.PUT_LINE('  * Set Discount : ' || (p_discount * 100) || '%');
    DBMS_OUTPUT.PUT_LINE(' ');

    DBMS_OUTPUT.PUT_LINE(' PACKAGE ITEMS ALLOCATION ');
    DBMS_OUTPUT.PUT_LINE('  * ' || p_item1_id || ' - ' || v_item1_name || ' (x' || p_qty1 || ')');
    DBMS_OUTPUT.PUT_LINE('  * ' || p_item2_id || ' - ' || v_item2_name || ' (x' || p_qty2 || ')');
    IF p_item3_id IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('  * ' || p_item3_id || ' - ' || v_item3_name || ' (x' || p_qty3 || ')');
    END IF;
    DBMS_OUTPUT.PUT_LINE(' ');

    DBMS_OUTPUT.PUT_LINE(' PACKAGE PRICING CALCULATION ');
    DBMS_OUTPUT.PUT_LINE(RPAD('  * Original Accumulated Price', 40) || ': RM ' || TO_CHAR(v_orig_price, 'FM99,990.90'));
    DBMS_OUTPUT.PUT_LINE(RPAD('  * Discount Applied', 40) || ': ' || (p_discount * 100) || '%');
    DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(RPAD(' >> FINAL DISCOUNTED PACKAGE PRICE', 40) || ': RM ' || TO_CHAR(v_disc_price, 'FM99,990.90'));
    DBMS_OUTPUT.PUT_LINE('====================================================================================================');
    DBMS_OUTPUT.PUT_LINE('* SUCCESS: New Package created and items linked successfully. *');
    DBMS_OUTPUT.PUT_LINE('====================================================================================================');

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('EXECUTION TERMINATED WITH EXCEPTION');
        DBMS_OUTPUT.PUT_LINE('ERROR at procedure execution:');
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
        DBMS_OUTPUT.PUT_LINE('====================================================================================================');
END;
/

SET DEFINE ON;
