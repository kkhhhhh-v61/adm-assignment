COLUMN OrderID FORMAT A7
COLUMN DeliveryID FORMAT A10
COLUMN DeliveryRiderID FORMAT A15
COLUMN OrderStatus FORMAT A11
COLUMN MAX(DeliveryID) FORMAT A15

SELECT MAX(DELIVERYID) FROM DELIVERY;

SELECT OrderID, DeliveryID, OrderStatus
FROM ORDERS
WHERE OrderID = 'O00596';

PROMPT CASE 6: VALID ASSIGNMENT (Standard / Tier 1 Rider)

SELECT O.OrderID, O.DeliveryID, D.DeliveryRiderID, O.OrderStatus
FROM ORDERS O

WHERE O.OrderID = 'O00596';
