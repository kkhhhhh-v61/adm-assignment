COLUMN OrderID FORMAT A7
COLUMN DeliveryID FORMAT A10
COLUMN DeliveryRiderID FORMAT A15
COLUMN OrderStatus FORMAT A11

SELECT O.OrderID, O.DeliveryID, D.DeliveryRiderID, O.OrderStatus
FROM ORDERS O
JOIN DELIVERY D ON O.DeliveryID = D.DeliveryID
WHERE D.DeliveryRiderID = 'DR10001' AND O.OrderStatus = 'Delivering';

SELECT O.OrderID, O.DeliveryID, D.DeliveryRiderID, O.OrderStatus
FROM ORDERS O

WHERE D.DeliveryRiderID = 'DR10002' AND O.OrderStatus = 'Delivering';
