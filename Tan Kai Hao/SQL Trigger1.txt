CL SCR

CREATE OR REPLACE TRIGGER trg_limit_rider_workload
BEFORE INSERT ON DELIVERY
FOR EACH ROW
DECLARE
    PRAGMA AUTONOMOUS_TRANSACTION;
    v_active_count NUMBER;
    v_rider_rating DELIVERYRIDER.AverageRating%TYPE;
    v_max_limit    NUMBER;
BEGIN
    -- 1. Get current active deliveries
    SELECT COUNT(O.OrderID)
    INTO v_active_count
    FROM ORDERS O
    JOIN DELIVERY D ON O.DeliveryID = D.DeliveryID
    WHERE D.DeliveryRiderID = :NEW.DeliveryRiderID
    AND O.OrderStatus = 'Delivering';

    -- 2. Get Rider's Rating
    SELECT AverageRating 
    INTO v_rider_rating 
    FROM DELIVERYRIDER 
    WHERE DeliveryRiderID = :NEW.DeliveryRiderID;

    -- 3. Determine Dynamic Limit
    IF v_rider_rating >= 4.5 THEN
        v_max_limit := 5;
    ELSIF v_rider_rating >= 3.0 THEN
        v_max_limit := 3;
    ELSE
        v_max_limit := 1;
    END IF;

    -- 4. Validate
    IF v_active_count >= v_max_limit THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20010, 'Workload Limit Exceeded: Rider rating is ' || v_rider_rating || ' (Max allowed: ' || v_max_limit || ' deliveries).');
    END IF;
    
    COMMIT;
END;
/
