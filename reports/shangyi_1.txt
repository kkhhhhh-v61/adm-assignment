CL SCR
SET SERVEROUTPUT ON FORMAT WRAPPED;

CREATE OR REPLACE PROCEDURE generate_item_sales_report(
    p_start_date IN DATE,
    p_end_date   IN DATE
)
AS

    CURSOR c_restaurant_sales IS
        SELECT
            DENSE_RANK() OVER (ORDER BY SUM(OI.Quantity * OI.PriceEach) DESC) AS rank_no,
            R.RestaurantName,
            COUNT(DISTINCT O.OrderID) AS total_orders,
            SUM(OI.Quantity) AS total_qty,
            SUM(OI.Quantity * OI.PriceEach) AS revenue
        FROM RESTAURANT R
        JOIN ITEM I ON R.RestaurantID = I.RestaurantID
        JOIN ORDERITEM OI ON I.ItemID = OI.ItemID
        JOIN ORDERS O ON OI.OrderID = O.OrderID
        WHERE O.OrderStatus = 'Completed'
        AND TRUNC(O.OrderDate) BETWEEN TRUNC(p_start_date) AND TRUNC(p_end_date)
        GROUP BY R.RestaurantName
        ORDER BY revenue DESC;

    v_grand_orders NUMBER := 0;
    v_grand_qty NUMBER := 0;
    v_grand_revenue NUMBER := 0;

    v_top_revenue_restaurant VARCHAR2(120) := 'N/A';
    v_top_qty_restaurant VARCHAR2(120) := 'N/A';

    v_top_revenue NUMBER := -1;
    v_top_qty NUMBER := -1;

    v_percentage NUMBER;
    v_bar VARCHAR2(100);

    e_invalid_date EXCEPTION;
    e_no_data EXCEPTION;

    v_has_data BOOLEAN := FALSE;

BEGIN

    IF TRUNC(p_start_date) > TRUNC(p_end_date) THEN
        RAISE e_invalid_date;
    END IF;

    -- Calculate totals
    FOR r IN c_restaurant_sales LOOP

        v_has_data := TRUE;

        v_grand_orders := v_grand_orders + r.total_orders;
        v_grand_qty := v_grand_qty + r.total_qty;
        v_grand_revenue := v_grand_revenue + r.revenue;

        IF r.revenue > v_top_revenue THEN
            v_top_revenue := r.revenue;
            v_top_revenue_restaurant := r.RestaurantName;
        END IF;

        IF r.total_qty > v_top_qty THEN
            v_top_qty := r.total_qty;
            v_top_qty_restaurant := r.RestaurantName;
        END IF;

    END LOOP;

    IF NOT v_has_data THEN
        RAISE e_no_data;
    END IF;

    DBMS_OUTPUT.PUT_LINE('===========================================================================================================================');
    DBMS_OUTPUT.PUT_LINE('                                       RESTAURANT ITEM SALES COMPARISON REPORT');
    DBMS_OUTPUT.PUT_LINE('===========================================================================================================================');

    DBMS_OUTPUT.PUT_LINE('Period    : ' || TO_CHAR(p_start_date,'DD-MON-YYYY') ||
                         ' to ' || TO_CHAR(p_end_date,'DD-MON-YYYY'));

    DBMS_OUTPUT.PUT_LINE('Generated : ' || TO_CHAR(SYSDATE,'DD-MON-YYYY HH24:MI:SS'));

    DBMS_OUTPUT.PUT_LINE('---------------------------------------------------------------------------------------------------------------------------');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('RANK',6) ||
        RPAD('RESTAURANT',25) ||
        LPAD('ORDERS',10) ||
        LPAD('QTY SOLD',12) ||
        LPAD('AVG ITEM PRICE',18) ||
        LPAD('REVENUE (RM)',18) ||
        LPAD('REVENUE (%)',14) ||
        '   REVENUE SHARE'
    );

    DBMS_OUTPUT.PUT_LINE('---------------------------------------------------------------------------------------------------------------------------');

    FOR r IN c_restaurant_sales LOOP

        v_percentage := ROUND((r.revenue / v_grand_revenue) * 100, 2);

        -- Oracle-friendly bar (1 block ≈ 2%)
        v_bar := RPAD('#', ROUND(v_percentage/2), '#');

        DBMS_OUTPUT.PUT_LINE(
            RPAD(r.rank_no,6) ||
            RPAD(r.RestaurantName,25) ||
            LPAD(r.total_orders,10) ||
            LPAD(r.total_qty,12) ||
            LPAD(ROUND(r.revenue / NULLIF(r.total_qty,0),2),18) ||
            LPAD(TO_CHAR(r.revenue,'999,999,990.99'),18) ||
            LPAD(TO_CHAR(v_percentage,'990.00') || '%',14) ||
            '   ' || v_bar
        );

    END LOOP;

    DBMS_OUTPUT.PUT_LINE('===========================================================================================================================');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('GRAND TOTAL',31) ||
        LPAD(v_grand_orders,10) ||
        LPAD(v_grand_qty,12) ||
        LPAD('-',18) ||
        LPAD(TO_CHAR(v_grand_revenue,'999,999,990.99'),18) ||
        LPAD('100.00%',14)
    );

    DBMS_OUTPUT.PUT_LINE('===========================================================================================================================');

    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('=================================================== EXECUTIVE SUMMARY =====================================================');

    DBMS_OUTPUT.PUT_LINE('>> TOP RESTAURANT BY REVENUE : ' || v_top_revenue_restaurant ||
                         ' (RM ' || TO_CHAR(v_top_revenue,'999,999,990.99') || ')');

    DBMS_OUTPUT.PUT_LINE('>> TOP RESTAURANT BY QUANTITY SOLD : ' || v_top_qty_restaurant ||
                         ' (' || v_top_qty || ' ITEMS SOLD)');

    DBMS_OUTPUT.PUT_LINE('===========================================================================================================================');
    DBMS_OUTPUT.PUT_LINE('End of Report.');

EXCEPTION

    WHEN e_invalid_date THEN
        RAISE_APPLICATION_ERROR(-20001,'INVALID DATE RANGE: Start date cannot be later than end date.');

    WHEN e_no_data THEN
        DBMS_OUTPUT.PUT_LINE('>>> NO ITEM SALES DATA FOUND FOR THE GIVEN DATE RANGE <<<');

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Unexpected Error: ' || SQLERRM);

END;
