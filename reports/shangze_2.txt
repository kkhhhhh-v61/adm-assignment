CL SCR

SET SERVEROUTPUT ON FORMAT WRAPPED;
SET LINESIZE 160;

CREATE OR REPLACE PROCEDURE proc_revenue (
    p_start_date IN DATE,
    p_end_date   IN DATE
) AS
    -- Cursor: All packages sold
    CURSOR c_packages IS
        SELECT P.PackageID, P.PackageName, P.DiscountedPrice, 
               SUM(OP.Quantity) as TimesPkgSold
        FROM PACKAGE P
        JOIN ORDERPACKAGE OP ON P.PackageID = OP.PackageID
        JOIN ORDERS O ON OP.OrderID = O.OrderID
        WHERE O.OrderStatus = 'Completed'
          AND TRUNC(O.OrderDate) BETWEEN TRUNC(p_start_date) AND TRUNC(p_end_date)
        GROUP BY P.PackageID, P.PackageName, P.DiscountedPrice;

    -- Nested Cursor: Items inside the package
    CURSOR c_pkg_contents(v_pkg_id CHAR) IS
        SELECT PI.ItemID, PI.Quantity as QtyPerPkg, I.Price as UnitPrice
        FROM PACKAGEITEM PI
        JOIN ITEM I ON PI.ItemID = I.ItemID
        WHERE PI.PackageID = v_pkg_id;

    v_original_pkg_value NUMBER := 0;
    v_actual_pkg_revenue NUMBER := 0;
    v_total_indiv_sales  NUMBER := 0;
    v_indiv_item_rev     NUMBER := 0;
    
    v_diff_percent       NUMBER := 0;
    v_status             VARCHAR2(20);
    v_data_exists        BOOLEAN := FALSE;

BEGIN
    -- Header
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 150, '='));
    DBMS_OUTPUT.PUT_LINE(LPAD('PACKAGE REVENUE VS. INDEPENDENT ITEM SALES REPORT', 95));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 150, '='));
    DBMS_OUTPUT.PUT_LINE(
        RPAD('PACKAGE NAME', 30) || 
        LPAD('PKG REVENUE', 25) || 
        LPAD('INDIV. SALES REV', 25) || 
        LPAD('REV DIFF (%)', 20) || 
        LPAD('HIGHER YIELD', 25)
    );
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 150, '-'));

    FOR r_pkg IN c_packages LOOP
        v_data_exists        := TRUE;
        v_original_pkg_value := 0;
        v_total_indiv_sales  := 0;

        -- 1. Calculate Package Revenue based on Item components
        FOR r_item IN c_pkg_contents(r_pkg.PackageID) LOOP
            -- Sum of (Item Price * Qty in Package)
            v_original_pkg_value := v_original_pkg_value + (r_item.UnitPrice * r_item.QtyPerPkg);
            
            -- 2. Calculate Total Revenue of these items when sold INDIVIDUALLY (from ORDERITEM)
            SELECT NVL(SUM(Quantity * PriceEach), 0)
            INTO v_indiv_item_rev
            FROM ORDERITEM OI
            JOIN ORDERS O ON OI.OrderID = O.OrderID
            WHERE OI.ItemID = r_item.ItemID
              AND OI.ItemSetID IS NULL -- Ensuring it was sold individually, not in a set
              AND O.OrderStatus = 'Completed'
              AND TRUNC(O.OrderDate) BETWEEN TRUNC(p_start_date) AND TRUNC(p_end_date);
              
            v_total_indiv_sales := v_total_indiv_sales + v_indiv_item_rev;
        END LOOP;

        -- Actual Package Revenue = (Sum of Items - Discount) * Number of Packages Sold
        v_actual_pkg_revenue := (v_original_pkg_value - r_pkg.DiscountedPrice) * r_pkg.TimesPkgSold;

        -- 3. Calculate Percentage Difference
        IF v_total_indiv_sales > 0 THEN
            v_diff_percent := ((v_actual_pkg_revenue - v_total_indiv_sales) / v_total_indiv_sales) * 100;
        ELSE
            v_diff_percent := 100; -- If no indiv sales, package is 100% of the revenue
        END IF;

        -- 4. Determine which yielded more revenue
        IF v_actual_pkg_revenue > v_total_indiv_sales THEN
            v_status := 'PACKAGE';
        ELSE
            v_status := 'INDIVIDUAL';
        END IF;

        -- Output Row
        DBMS_OUTPUT.PUT_LINE(
            RPAD(r_pkg.PackageName, 30) || 
            LPAD('RM ' || TO_CHAR(v_actual_pkg_revenue, '999,990.99'), 25) || 
            LPAD('RM ' || TO_CHAR(v_total_indiv_sales, '999,990.99'), 25) || 
            LPAD(TO_CHAR(v_diff_percent, '990.99') || '%', 20) || 
            LPAD(v_status, 25)
        );
    END LOOP;

    -- Conclusion
    IF v_data_exists THEN
        DBMS_OUTPUT.PUT_LINE(RPAD('=', 150, '='));
        DBMS_OUTPUT.PUT_LINE('CONCLUSION:');
        DBMS_OUTPUT.PUT_LINE('This report compares the revenue generated by selling items bundled in packages');
        DBMS_OUTPUT.PUT_LINE('versus the revenue generated by those same items sold as standalone individual units.');
        DBMS_OUTPUT.PUT_LINE('If "INDIVIDUAL" shows a higher yield, consider increasing package pricing or promotion.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('No package sales found for the selected dates.');
    END IF;
    
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 150, '='));

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error in report: ' || SQLERRM);
END;
