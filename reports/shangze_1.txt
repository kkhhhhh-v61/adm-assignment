CL SCR
SET SERVEROUTPUT ON FORMAT WRAPPED;

CREATE OR REPLACE PROCEDURE generate_deliverytype_report (
    p_start_date IN DATE,
    p_end_date   IN DATE
) AS

    -- Cursor to get delivery types
    CURSOR c_deliverytype IS
        SELECT DISTINCT DT.DeliveryName, DT.DeliveryTypeID
        FROM DELIVERYTYPE DT
        JOIN DELIVERY D ON DT.DeliveryTypeID = D.DeliveryTypeID
        JOIN ORDERS O ON O.DeliveryID = D.DeliveryID
        WHERE O.OrderStatus = 'Completed'
        AND TRUNC(O.OrderDate) BETWEEN TRUNC(p_start_date) AND TRUNC(p_end_date)
        ORDER BY DT.DeliveryName;

    -- Nested cursor to calculate orders and revenue
    CURSOR c_orders(v_deliverytype_id CHAR) IS
        SELECT COUNT(O.OrderID) AS OrderVolume,
               SUM(O.TotalAmount) AS TotalRevenue
        FROM ORDERS O
        JOIN DELIVERY D ON O.DeliveryID = D.DeliveryID
        WHERE O.OrderStatus = 'Completed'
        AND D.DeliveryTypeID = v_deliverytype_id
        AND TRUNC(O.OrderDate) BETWEEN TRUNC(p_start_date) AND TRUNC(p_end_date);

    -- Record structure
    TYPE t_delivery_data IS RECORD (
        delivery_name VARCHAR2(100),
        order_volume  NUMBER,
        revenue       NUMBER
    );

    -- Collection
    TYPE t_delivery_tab IS TABLE OF t_delivery_data INDEX BY PLS_INTEGER;
    v_data t_delivery_tab;

    v_index PLS_INTEGER := 0;

    v_grand_orders NUMBER := 0;
    v_grand_revenue NUMBER := 0;

    -- Top revenue
    v_top_deliverytype VARCHAR2(100) := 'N/A';
    v_top_revenue NUMBER := -1;

    -- Top orders
    v_top_orders_delivery VARCHAR2(100) := 'N/A';
    v_top_orders NUMBER := -1;

    -- Ranking
    v_rank NUMBER;

    -- Exceptions
    e_invalid_date EXCEPTION;
    e_no_data EXCEPTION;

    v_data_exists BOOLEAN := FALSE;

BEGIN

    IF TRUNC(p_start_date) > TRUNC(p_end_date) THEN
        RAISE e_invalid_date;
    END IF;

    -- Collect data
    FOR r_dt IN c_deliverytype LOOP

        v_data_exists := TRUE;
        v_index := v_index + 1;

        v_data(v_index).delivery_name := r_dt.DeliveryName;
        v_data(v_index).order_volume := 0;
        v_data(v_index).revenue := 0;

        FOR r_order IN c_orders(r_dt.DeliveryTypeID) LOOP
            v_data(v_index).order_volume := NVL(r_order.OrderVolume,0);
            v_data(v_index).revenue := NVL(r_order.TotalRevenue,0);
        END LOOP;

        v_grand_orders := v_grand_orders + v_data(v_index).order_volume;
        v_grand_revenue := v_grand_revenue + v_data(v_index).revenue;

        -- Top revenue
        IF v_data(v_index).revenue > v_top_revenue THEN
            v_top_revenue := v_data(v_index).revenue;
            v_top_deliverytype := r_dt.DeliveryName;
        END IF;

        -- Top orders
        IF v_data(v_index).order_volume > v_top_orders THEN
            v_top_orders := v_data(v_index).order_volume;
            v_top_orders_delivery := r_dt.DeliveryName;
        END IF;

    END LOOP;

    IF NOT v_data_exists THEN
        RAISE e_no_data;
    END IF;

    -- Report Header
    DBMS_OUTPUT.PUT_LINE('===============================================================================================');
    DBMS_OUTPUT.PUT_LINE('                  DELIVERY TYPE REVENUE AND ORDER PERFORMANCE REPORT');
    DBMS_OUTPUT.PUT_LINE('===============================================================================================');

    DBMS_OUTPUT.PUT_LINE('Period    : ' || TO_CHAR(p_start_date,'DD-MON-YYYY') ||
                         ' to ' || TO_CHAR(p_end_date,'DD-MON-YYYY'));

    DBMS_OUTPUT.PUT_LINE('Generated : ' || TO_CHAR(SYSDATE,'DD-MON-YYYY HH24:MI:SS'));

    DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------------------------------');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('DELIVERY TYPE',25) ||
        LPAD('ORDERS',10) ||
        LPAD('ORDER (%)',12) ||
        LPAD('REVENUE (RM)',15) ||
        LPAD('REVENUE (%)',15) ||
        LPAD('AVG ORDER (RM)',18)
    );

    DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------------------------------');

    -- Print each row
    FOR i IN 1..v_index LOOP

        DBMS_OUTPUT.PUT_LINE(

            RPAD(v_data(i).delivery_name,25) ||

            LPAD(v_data(i).order_volume,10) ||

            LPAD(ROUND((v_data(i).order_volume / NULLIF(v_grand_orders,0))*100,2)||'%',12) ||

            LPAD(TO_CHAR(v_data(i).revenue,'999,999,990.99'),15) ||

            LPAD(ROUND((v_data(i).revenue / NULLIF(v_grand_revenue,0))*100,2)||'%',15) ||

            LPAD(
                TO_CHAR(
                    v_data(i).revenue / NULLIF(v_data(i).order_volume,0),
                    '999,999,990.99'
                ),18
            )

        );

    END LOOP;

    -- Grand total
    DBMS_OUTPUT.PUT_LINE('===============================================================================================');

    DBMS_OUTPUT.PUT_LINE(
        RPAD('GRAND TOTAL',25) ||
        LPAD(v_grand_orders,10) ||
        LPAD('100%',12) ||
        LPAD(TO_CHAR(v_grand_revenue,'999,999,990.99'),15) ||
        LPAD('100%',15)
    );

    DBMS_OUTPUT.PUT_LINE('===============================================================================================');

    -- Executive summary
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('===================================== EXECUTIVE SUMMARY =======================================');

    DBMS_OUTPUT.PUT_LINE('>> TOP DELIVERY TYPE BY REVENUE : ' ||
                         UPPER(v_top_deliverytype) ||
                         ' (RM ' || TRIM(TO_CHAR(v_top_revenue,'999,999,990.99')) || ')');

    DBMS_OUTPUT.PUT_LINE('>> MOST FREQUENT DELIVERY TYPE  : ' ||
                         UPPER(v_top_orders_delivery) ||
                         ' (' || v_top_orders || ' ORDERS)');


    DBMS_OUTPUT.PUT_LINE('===============================================================================================');
    DBMS_OUTPUT.PUT_LINE('End of Report.');

EXCEPTION

    WHEN e_invalid_date THEN
        RAISE_APPLICATION_ERROR(-20001,'INVALID DATE RANGE: Start Date cannot be later than End Date.');

    WHEN e_no_data THEN
        DBMS_OUTPUT.PUT_LINE('>>> NO COMPLETED ORDERS FOUND FOR THIS DATE RANGE <<<');

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Unexpected Error: ' || SQLERRM);

END;
