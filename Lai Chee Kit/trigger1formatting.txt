SET SERVEROUTPUT ON;
SET LINESIZE 150;

PROMPT >>> ATTEMPTING NEW ORDER FOR M00001... <<<

DECLARE
    v_error_msg VARCHAR2(4000);
BEGIN
    -- Member M00001 tries to order food again!
    INSERT INTO ORDERS (OrderID, MemberID, DeliveryID, MemberVoucherID, PickupType, OrderDate, OrderTime, TotalAmount, OrderStatus) 
    VALUES ('O88882', 'M00001', NULL, NULL, 'Pickup', SYSDATE, '13:00', 15.00, 'Pending');
    
    DBMS_OUTPUT.PUT_LINE('✅ SUCCESS: Order placed successfully.');
    
EXCEPTION
    WHEN OTHERS THEN
        -- 1. Grab the raw error message
        v_error_msg := SQLERRM;
        
        -- 2. Remove the "ORA-20001: " prefix
        v_error_msg := REPLACE(v_error_msg, 'ORA-20001: ', '');
        
        -- 3. Chop off everything from "ORA-06512" onwards
        IF INSTR(v_error_msg, 'ORA-06512') > 0 THEN
            v_error_msg := SUBSTR(v_error_msg, 1, INSTR(v_error_msg, 'ORA-06512') - 2);
        END IF;

        -- 4. Print the perfectly clean UI
        DBMS_OUTPUT.PUT_LINE('==================================================');
        DBMS_OUTPUT.PUT_LINE('❌ TRANSACTION BLOCKED BY DATABASE TRIGGER');
        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('Reason: ' || TRIM(REPLACE(v_error_msg, CHR(10), '')));
        DBMS_OUTPUT.PUT_LINE('==================================================');
END;
