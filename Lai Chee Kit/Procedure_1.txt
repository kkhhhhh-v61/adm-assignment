CREATE OR REPLACE PROCEDURE PR_MANAGE_SUBSCRIPTION (
    p_action      IN VARCHAR2, 
    p_member_id   IN MEMBER.MemberID%TYPE,
    p_plan_id     IN SUBSCRIPTIONPLAN.SubscriptionPlanID%TYPE DEFAULT NULL
) AS
    v_member_name    MEMBER.FullName%TYPE;
    v_plan_rec       SUBSCRIPTIONPLAN%ROWTYPE;
    v_old_end_date   DATE;
    v_final_amount   NUMBER(6,2);
    v_discount_rate  NUMBER(3,2) := 0;
    v_new_sub_id     CHAR(6);
    v_new_pay_id     CHAR(7);
    v_new_start      DATE;
    v_new_end        DATE;
    
    v_active_count   NUMBER := 0;
    v_past_count     NUMBER := 0;
    v_last_expiry    DATE;

    e_invalid_action EXCEPTION;
    e_invalid_member EXCEPTION;
    e_invalid_plan   EXCEPTION;
    e_no_active_sub  EXCEPTION;
BEGIN
    SAVEPOINT start_process;

    -- 1. VALIDATE THE ACTION
    IF UPPER(p_action) NOT IN ('EXTEND', 'CANCEL') THEN
        RAISE e_invalid_action;
    END IF;

    -- 2. VALIDATE THE MEMBER
    BEGIN
        SELECT FullName INTO v_member_name FROM MEMBER WHERE MemberID = p_member_id;
    EXCEPTION WHEN NO_DATA_FOUND THEN RAISE e_invalid_member; END;

    -- ==========================================
    -- ACTION: EXTEND (RENEWALS)
    -- ==========================================
    IF UPPER(p_action) = 'EXTEND' THEN
        -- Validate Plan
        BEGIN
            SELECT * INTO v_plan_rec FROM SUBSCRIPTIONPLAN WHERE SubscriptionPlanID = p_plan_id;
        EXCEPTION WHEN NO_DATA_FOUND THEN RAISE e_invalid_plan; END;

        -- Check for existing active subscriptions to apply discount
        SELECT COUNT(*) INTO v_active_count FROM SUBSCRIPTION 
        WHERE MemberID = p_member_id AND Status = 'Active' AND EndDate >= SYSDATE;

        IF v_active_count > 0 THEN
            DBMS_OUTPUT.PUT_LINE('[INFO] Early renewal detected (Active Plan). 5% Discount applied!');
            v_discount_rate := 0.05;
            SELECT MAX(EndDate) INTO v_old_end_date FROM SUBSCRIPTION 
            WHERE MemberID = p_member_id AND Status = 'Active';
        ELSE
            -- EXPIRED VIP FEATURE: Check if they have an old expired/cancelled plan
            SELECT COUNT(*) INTO v_past_count FROM SUBSCRIPTION WHERE MemberID = p_member_id;
            
            IF v_past_count > 0 THEN
                SELECT MAX(EndDate) INTO v_last_expiry FROM SUBSCRIPTION WHERE MemberID = p_member_id;
                DBMS_OUTPUT.PUT_LINE('[INFO] Standard Renewal. Previous VIP expired on: ' || TO_CHAR(v_last_expiry, 'DD-MON-YYYY'));
            ELSE
                DBMS_OUTPUT.PUT_LINE('[INFO] Standard Renewal (First-Time VIP). No early-bird discount applied.');
            END IF;
            
            v_old_end_date := SYSDATE;
        END IF;

        -- Apply Math
        v_final_amount := v_plan_rec.Price * (1.00 - v_discount_rate);
        v_new_start := v_old_end_date;
        v_new_end := v_new_start + v_plan_rec.DurationInDays;

        -- Generate Auto-IDs (Starting from 82000)
        SELECT 'S' || LPAD(NVL(MAX(TO_NUMBER(SUBSTR(SubscriptionID, 2))), 82000) + 1, 5, '0') 
        INTO v_new_sub_id FROM SUBSCRIPTION;

        SELECT 'PM' || LPAD(NVL(MAX(TO_NUMBER(SUBSTR(PaymentID, 3))), 82000) + 1, 5, '0') 
        INTO v_new_pay_id FROM PAYMENT;

        -- Insert Records
        INSERT INTO SUBSCRIPTION (SubscriptionID, MemberID, SubscriptionPlanID, StartDate, EndDate, Status)
        VALUES (v_new_sub_id, p_member_id, p_plan_id, v_new_start, v_new_end, 'Active');

        INSERT INTO PAYMENT (PaymentID, OrderID, SubscriptionID, PaymentMethod, PaymentStatus)
        VALUES (v_new_pay_id, NULL, v_new_sub_id, 'Credit Card', 'Completed');

        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('[SUCCESS] Subscription Extended.');
        DBMS_OUTPUT.PUT_LINE('Member          : ' || v_member_name || ' (' || p_member_id || ')');
        DBMS_OUTPUT.PUT_LINE('Plan Selected   : ' || v_plan_rec.PlanName || ' (' || p_plan_id || ')');
        DBMS_OUTPUT.PUT_LINE('New Sub ID      : ' || v_new_sub_id);
        DBMS_OUTPUT.PUT_LINE('Amount Paid     : RM ' || TO_CHAR(v_final_amount, '999.99'));
        DBMS_OUTPUT.PUT_LINE('Valid Period    : ' || TO_CHAR(v_new_start, 'DD-MON-YYYY') || ' to ' || TO_CHAR(v_new_end, 'DD-MON-YYYY'));
        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');

    -- ==========================================
    -- ACTION: CANCEL
    -- ==========================================
    ELSIF UPPER(p_action) = 'CANCEL' THEN
        SELECT COUNT(*) INTO v_active_count FROM SUBSCRIPTION 
        WHERE MemberID = p_member_id AND Status = 'Active';

        IF v_active_count = 0 THEN
            RAISE e_no_active_sub;
        END IF;

        UPDATE SUBSCRIPTION 
        SET Status = 'Cancelled', 
            EndDate = CASE 
                        WHEN StartDate > SYSDATE THEN StartDate 
                        ELSE SYSDATE 
                      END
        WHERE MemberID = p_member_id AND Status = 'Active';

        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('[SUCCESS] Subscription Cancelled.');
        DBMS_OUTPUT.PUT_LINE('Member          : ' || v_member_name || ' (' || p_member_id || ')');
        DBMS_OUTPUT.PUT_LINE('Action          : All active plans successfully cancelled.');
        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
    END IF;

    COMMIT;

EXCEPTION
    WHEN e_invalid_action THEN
        ROLLBACK TO start_process;
        DBMS_OUTPUT.PUT_LINE('ERROR: Action must be either ''EXTEND'' or ''CANCEL''.');
    WHEN e_invalid_member THEN
        ROLLBACK TO start_process;
        DBMS_OUTPUT.PUT_LINE('ERROR: Member ID (' || p_member_id || ') does not exist.');
    WHEN e_invalid_plan THEN
        ROLLBACK TO start_process;
        DBMS_OUTPUT.PUT_LINE('ERROR: Subscription Plan ID (' || p_plan_id || ') does not exist.');
    WHEN e_no_active_sub THEN
        ROLLBACK TO start_process;
        DBMS_OUTPUT.PUT_LINE('ERROR: Member has no active subscriptions to cancel.');
    WHEN OTHERS THEN
        ROLLBACK TO start_process;
        DBMS_OUTPUT.PUT_LINE('An unexpected error occurred: ' || SQLERRM);
END;
