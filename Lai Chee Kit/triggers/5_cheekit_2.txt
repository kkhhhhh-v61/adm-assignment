CREATE OR REPLACE TRIGGER TRG_MASTER_FINANCE_GATEWAY
BEFORE INSERT OR UPDATE ON PAYMENT
FOR EACH ROW
DECLARE
    v_order_status ORDERS.OrderStatus%TYPE;
    v_order_total  ORDERS.TotalAmount%TYPE;
    v_pickup_type  ORDERS.PickupType%TYPE;
BEGIN
    -- ========================================================================
    -- RULE 1: IMMUTABILITY OF COMPLETED FINANCIAL TRANSACTIONS
    -- Prevent tampering with successful transactions.
    -- ========================================================================
    IF UPDATING AND :OLD.PaymentStatus = 'Completed' AND :NEW.PaymentStatus != 'Completed' THEN
        RAISE_APPLICATION_ERROR(-20020, 'Audit Error: A Completed payment is locked and cannot be reverted to ' || :NEW.PaymentStatus || '.');
    END IF;

    -- ========================================================================
    -- RULE 2: SUBSCRIPTION PAYMENT LOGIC
    -- ========================================================================
    IF :NEW.SubscriptionID IS NOT NULL THEN
        IF :NEW.PaymentMethod = 'Cash on Delivery' THEN
            RAISE_APPLICATION_ERROR(-20021, 'Payment Error: Digital Subscriptions do not support Cash on Delivery. Please use Card or E-Wallet.');
        END IF;
    END IF;

    -- ========================================================================
    -- RULES 3, 4, and 5: ORDER PAYMENT LOGIC (Cross-Table Validations)
    -- ========================================================================
    IF :NEW.OrderID IS NOT NULL THEN
        
               BEGIN
            SELECT OrderStatus, TotalAmount, PickupType
            INTO v_order_status, v_order_total, v_pickup_type
            FROM ORDERS
            WHERE TRIM(OrderID) = TRIM(:NEW.OrderID);
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                RAISE_APPLICATION_ERROR(-20026, 'System Error: Order ID [' || :NEW.OrderID || '] does not exist in the ORDERS table.');
        END;

        IF v_order_status = 'Cancelled' AND :NEW.PaymentStatus IN ('Pending', 'Completed') THEN
            RAISE_APPLICATION_ERROR(-20022, 'Payment Error: Cannot process payments for an order that has been Cancelled.');
        END IF;

        IF :NEW.PaymentMethod = 'Cash on Delivery' THEN
            
            IF v_pickup_type = 'Pickup' THEN
                RAISE_APPLICATION_ERROR(-20023, 'Payment Error: Cash on Delivery is strictly for Deliveries. Please use E-Wallet or Card for Pickups.');
            END IF;

            IF v_order_total > 200.00 THEN
                RAISE_APPLICATION_ERROR(-20024, 'Security Error: Cash on Delivery is disabled for orders exceeding RM 200.00 for rider safety. Please use a digital payment method.');
            END IF;

            IF :NEW.PaymentStatus = 'Completed' AND v_order_status = 'Pending' THEN
                RAISE_APPLICATION_ERROR(-20025, 'Process Error: Cash on Delivery cannot be marked Completed while the OrderStatus is still Pending.');
            END IF;
            
        END IF;
    END IF;
END;
