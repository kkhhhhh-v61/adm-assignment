CREATE OR REPLACE TRIGGER TRG_ENFORCE_DEBT_LOCK
BEFORE INSERT ON ORDERS
FOR EACH ROW
DECLARE
       PRAGMA AUTONOMOUS_TRANSACTION;
    v_debt_count NUMBER;
BEGIN
        SELECT COUNT(*)
    INTO v_debt_count
    FROM ORDERS o
    JOIN PAYMENT p ON o.OrderID = p.OrderID
    WHERE o.MemberID = :NEW.MemberID
    AND p.PaymentStatus IN ('Pending', 'Failed');

       IF v_debt_count > 0 THEN
        -- We must commit the autonomous transaction before raising an error
        COMMIT; 
        RAISE_APPLICATION_ERROR(-20001, 'ORDER BLOCKED: You have an outstanding or failed payment. Please settle your debt before placing a new order.');
    END IF;

        COMMIT; 
END;
/
