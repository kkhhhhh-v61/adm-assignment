CLEAR SCREEN;

SET LINESIZE 100;
SET PAGESIZE 50000;

TTITLE LEFT '             ========================================================' SKIP 1 -
       LEFT '              PEAK HOUR ORDER VOLUME AND DELIVERY BOTTLENECK SUMMARY' SKIP 1 -
       LEFT '             ========================================================' SKIP 2

COLUMN "Peak Hour" FORMAT A22;
COLUMN "Delivery Company" FORMAT A20;
COLUMN "Total Orders" FORMAT A12;
COLUMN "Avg Wait (Min)" FORMAT A14;
COLUMN "Performance Status" FORMAT A25;

WITH WaitTimeData AS (
    SELECT 
        TO_CHAR(TO_DATE(o.OrderTime, 'HH24:MI'), 'HH24') || ':00' AS PeakHour,
        dc.CompanyName AS DeliveryCompany,
        o.OrderID,
        MOD((TO_DATE(d.DeliveryTime, 'HH24:MI') - TO_DATE(o.OrderTime, 'HH24:MI')) * 24 * 60 + 1440, 1440) AS WaitTimeMins
    FROM 
        ORDERS o
    JOIN 
        DELIVERY d ON o.DeliveryID = d.DeliveryID
    JOIN 
        DELIVERYRIDER dr ON d.DeliveryRiderID = dr.DeliveryRiderID
    JOIN 
        DELIVERYCOMPANY dc ON dr.DeliveryCompanyID = dc.DeliveryCompanyID
    WHERE 
        o.OrderStatus = 'Completed'
),
MainData AS (
    SELECT 
        PeakHour AS PeakHour,
        DeliveryCompany AS DeliveryCompany,
        LPAD(TO_CHAR(COUNT(OrderID)), 12) AS TotalOrders,
        LPAD(TO_CHAR(ROUND(AVG(WaitTimeMins), 2), 'FM990.00'), 14) AS AvgWaitMins,
        CASE 
            WHEN ROUND(AVG(WaitTimeMins), 2) >= 120 THEN '🚨 CRITICAL BOTTLENECK'
            WHEN ROUND(AVG(WaitTimeMins), 2) >= 60  THEN '⚠️ HIGH TRAFFIC'
            ELSE '✅ NORMAL SPEED'
        END AS PerfStatus,
        1 AS SortOrder 
    FROM WaitTimeData
    GROUP BY PeakHour, DeliveryCompany
),
TotalData AS (
    SELECT 
        '=== SYSTEM AVERAGE ===' AS PeakHour,
        'ALL COMPANIES' AS DeliveryCompany,
        LPAD(TO_CHAR(COUNT(OrderID)), 12) AS TotalOrders,
        LPAD(TO_CHAR(ROUND(AVG(WaitTimeMins), 2), 'FM990.00'), 14) AS AvgWaitMins,
        CASE 
            WHEN ROUND(AVG(WaitTimeMins), 2) >= 120 THEN '🚨 CRITICAL BOTTLENECK'
            WHEN ROUND(AVG(WaitTimeMins), 2) >= 60  THEN '⚠️ HIGH TRAFFIC'
            ELSE '✅ NORMAL SPEED'
        END AS PerfStatus,
        3 AS SortOrder 
    FROM WaitTimeData
),
Separator AS (
    SELECT 
        '----------------------' AS PeakHour,
        '--------------------' AS DeliveryCompany,
        '------------' AS TotalOrders,
        '--------------' AS AvgWaitMins,
        '-------------------------' AS PerfStatus,
        2 AS SortOrder 
    FROM DUAL
)
SELECT 
    PeakHour AS "Peak Hour",
    DeliveryCompany AS "Delivery Company",
    TotalOrders AS "Total Orders",
    AvgWaitMins AS "Avg Wait (Min)",
    PerfStatus AS "Performance Status"
FROM (
    SELECT * FROM MainData
    UNION ALL
    SELECT * FROM Separator
    UNION ALL
    SELECT * FROM TotalData
)
ORDER BY 
    SortOrder ASC, 
    PeakHour DESC;

TTITLE OFF;
