CL SCR
SET SERVEROUTPUT ON FORMAT WRAPPED;

CREATE OR REPLACE PROCEDURE calculate_update_order_total (
p_order_id IN CHAR
)
AS
-- totals
v_item_total        NUMBER(10,2) := 0;
v_package_total     NUMBER(10,2) := 0;
v_delivery_fee      NUMBER(10,2) := 0;
v_subtotal          NUMBER(10,2) := 0;
v_final_total       NUMBER(10,2) := 0;


-- order info
v_pickup_type       ORDERS.PickupType%TYPE;
v_delivery_id       ORDERS.DeliveryID%TYPE;
v_member_voucher_id ORDERS.MemberVoucherID%TYPE;
v_order_status      ORDERS.OrderStatus%TYPE;

-- discount info
v_discount          ITEMSET.Discount%TYPE;

-- voucher variables
v_discount_percent  VOUCHER.DiscountPercentage%TYPE;
v_discount_amount   VOUCHER.DiscountAmount%TYPE;
v_min_spend         VOUCHER.MinSpendAmount%TYPE;
v_max_discount      VOUCHER.MaxDiscountAmount%TYPE;
v_voucher_discount  NUMBER(10,2) := 0;
v_expiry_date       DATE;

-- validation helpers
v_count             NUMBER := 0;


BEGIN


----------------------------------------------------
-- VALIDATION 1 : Order ID format
----------------------------------------------------
IF NOT REGEXP_LIKE(p_order_id,'^O[0-9]{5}$') THEN
    RAISE_APPLICATION_ERROR(-20601,'Invalid Order ID format.');
END IF;

----------------------------------------------------
-- VALIDATION 2 : Order exists
----------------------------------------------------
SELECT COUNT(*)
INTO v_count
FROM ORDERS
WHERE OrderID = p_order_id;

IF v_count = 0 THEN
    RAISE_APPLICATION_ERROR(-20602,'Order does not exist.');
END IF;

----------------------------------------------------
-- GET ORDER INFORMATION
----------------------------------------------------
SELECT PickupType, DeliveryID, MemberVoucherID, OrderStatus
INTO v_pickup_type, v_delivery_id, v_member_voucher_id, v_order_status
FROM ORDERS
WHERE OrderID = p_order_id;

----------------------------------------------------
-- VALIDATION 3 : Cannot modify completed/cancelled
----------------------------------------------------
IF v_order_status IN ('Completed','Cancelled') THEN
    RAISE_APPLICATION_ERROR(-20603,'Cannot modify completed or cancelled orders.');
END IF;

----------------------------------------------------
-- VALIDATION 4 : Order must contain item/package
----------------------------------------------------
SELECT COUNT(*)
INTO v_count
FROM (
    SELECT ItemID FROM ORDERITEM WHERE OrderID = p_order_id
    UNION ALL
    SELECT PackageID FROM ORDERPACKAGE WHERE OrderID = p_order_id
);

IF v_count = 0 THEN
    RAISE_APPLICATION_ERROR(-20604,'Order contains no items or packages.');
END IF;

----------------------------------------------------
-- VALIDATION 5 : Delivery logic
----------------------------------------------------
IF v_pickup_type = 'Delivery' AND v_delivery_id IS NULL THEN
    RAISE_APPLICATION_ERROR(-20605,'Delivery order must have delivery record.');
END IF;

----------------------------------------------------
-- VALIDATION 6 : Prevent recalculation after payment
----------------------------------------------------
SELECT COUNT(*)
INTO v_count
FROM PAYMENT
WHERE OrderID = p_order_id
AND PaymentStatus = 'Completed';

IF v_count > 0 THEN
    RAISE_APPLICATION_ERROR(-20606,'Order already paid. Cannot recalculate.');
END IF;


DBMS_OUTPUT.PUT_LINE('===========================================================================');
DBMS_OUTPUT.PUT_LINE('                         ORDER PAYMENT CALCULATION       ');
DBMS_OUTPUT.PUT_LINE('===========================================================================');
DBMS_OUTPUT.PUT_LINE('Order ID : ' || p_order_id);
DBMS_OUTPUT.PUT_LINE('---------------------------------------------------------------------------');


----------------------------------------------------
-- PROCESS ORDER ITEMS
----------------------------------------------------
FOR r IN (
    SELECT oi.ItemID, oi.Quantity, oi.PriceEach, oi.ItemSetID, i.ItemName
    FROM ORDERITEM oi
    JOIN ITEM i ON oi.ItemID = i.ItemID
    WHERE oi.OrderID = p_order_id
)
LOOP
    IF r.ItemSetID IS NOT NULL THEN

        SELECT Discount
        INTO v_discount
        FROM ITEMSET
        WHERE ItemSetID = r.ItemSetID;

        ------------------------------------------------
        -- VALIDATION 7 : ItemSet discount range
        ------------------------------------------------
        IF v_discount < 0 OR v_discount > 1 THEN
            RAISE_APPLICATION_ERROR(-20607,'Invalid item set discount.');
        END IF;

        v_item_total := v_item_total + (r.Quantity * r.PriceEach * (1 - v_discount));

        DBMS_OUTPUT.PUT_LINE(
            RPAD(r.ItemName || ' (Set Discount)',60) ||
            ' RM ' ||
            TO_CHAR(r.Quantity * r.PriceEach * (1 - v_discount),'9990.00')
        );

    ELSE
        v_item_total := v_item_total + (r.Quantity * r.PriceEach);

        DBMS_OUTPUT.PUT_LINE(
            RPAD(r.ItemName || ' x' || r.Quantity,60) ||
            ' RM ' ||
            TO_CHAR(r.Quantity * r.PriceEach,'9990.00')
        );
    END IF;

END LOOP;


----------------------------------------------------
-- PROCESS PACKAGES
----------------------------------------------------
FOR p IN (
    SELECT op.Quantity, op.PriceEach, pk.PackageName
    FROM ORDERPACKAGE op
    JOIN PACKAGE pk ON op.PackageID = pk.PackageID
    WHERE op.OrderID = p_order_id
)
LOOP
    v_package_total := v_package_total + (p.Quantity * p.PriceEach);

    DBMS_OUTPUT.PUT_LINE(
        RPAD(p.PackageName,60) ||
        ' RM ' ||
        TO_CHAR(p.Quantity * p.PriceEach,'9990.00')
    );
END LOOP;


----------------------------------------------------
-- DELIVERY FEE
----------------------------------------------------
IF v_pickup_type = 'Delivery' AND v_delivery_id IS NOT NULL THEN

    SELECT DiscountedDeliveryPrice
    INTO v_delivery_fee
    FROM DELIVERY
    WHERE DeliveryID = v_delivery_id;

    ------------------------------------------------
    -- VALIDATION 8 : Delivery fee check
    ------------------------------------------------
    IF v_delivery_fee < 0 THEN
        RAISE_APPLICATION_ERROR(-20608,'Invalid delivery fee.');
    END IF;

    DBMS_OUTPUT.PUT_LINE(
        RPAD('Delivery Fee',60) ||
        ' RM ' ||
        TO_CHAR(v_delivery_fee,'9990.00')
    );

END IF;


----------------------------------------------------
-- SUBTOTAL
----------------------------------------------------
v_subtotal := v_item_total + v_package_total + v_delivery_fee;


----------------------------------------------------
-- VOUCHER CALCULATION
----------------------------------------------------
IF v_member_voucher_id IS NOT NULL THEN

    BEGIN

        SELECT v.DiscountPercentage,
               v.DiscountAmount,
               v.MinSpendAmount,
               v.MaxDiscountAmount,
               v.ExpiryDate
        INTO v_discount_percent,
             v_discount_amount,
             v_min_spend,
             v_max_discount,
             v_expiry_date
        FROM MEMBERVOUCHER mv
        JOIN VOUCHER v ON mv.VoucherID = v.VoucherID
        WHERE mv.MemberVoucherID = v_member_voucher_id;

         v_voucher_discount := v_subtotal * v_discount_percent;


        DBMS_OUTPUT.PUT_LINE(
            RPAD('Voucher Discount',60) ||
            ' RM ' ||
            TO_CHAR(v_voucher_discount,'9990.00')
        );

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE(
                RPAD('Voucher Discount',60) ||
                ' RM ' ||
                LPAD('0.00',7)
            );
    END;

ELSE
    DBMS_OUTPUT.PUT_LINE(
        RPAD('Voucher Discount',60) ||
        ' RM ' ||
        LPAD('0.00',7)
    );
END IF;


----------------------------------------------------
-- FINAL TOTAL
----------------------------------------------------
v_final_total := v_subtotal - v_voucher_discount;

----------------------------------------------------
-- VALIDATION 11 : Prevent negative total
----------------------------------------------------
IF v_final_total < 0 THEN
    RAISE_APPLICATION_ERROR(-20610,'Final total cannot be negative.');
END IF;

----------------------------------------------------
-- VALIDATION 12 : Prevent extremely large orders
----------------------------------------------------
IF v_final_total > 10000 THEN
    RAISE_APPLICATION_ERROR(-20611,'Order amount exceeds system limit.');
END IF;


DBMS_OUTPUT.PUT_LINE('---------------------------------------------------------------------------');
DBMS_OUTPUT.PUT_LINE(
    RPAD('TOTAL',60) ||
    ' RM ' ||
    TO_CHAR(v_final_total,'9990.00')
);
DBMS_OUTPUT.PUT_LINE('===========================================================================');


----------------------------------------------------
-- UPDATE ORDER
----------------------------------------------------
UPDATE ORDERS
SET TotalAmount = v_final_total
WHERE OrderID = p_order_id;

COMMIT;

DBMS_OUTPUT.PUT_LINE('ORDER RECORD UPDATED');


EXCEPTION


WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('ERROR: ' || SQLERRM);
    ROLLBACK;


END;
