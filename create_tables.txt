CREATE TABLE ADDRESS (
  AddressID CHAR(6) NOT NULL,
  AddressLine VARCHAR2(255) NOT NULL,
  City VARCHAR2(100) NOT NULL,
  State VARCHAR2(100) NOT NULL,
  Postcode CHAR(5) NOT NULL,
  CONSTRAINT pk_addr_addressid PRIMARY KEY (AddressID),
  CONSTRAINT chk_addr_addressid CHECK (REGEXP_LIKE(AddressID, '^A[0-9]{5}$')),
  CONSTRAINT chk_addr_postcode CHECK (REGEXP_LIKE(Postcode, '^[0-9]{5}$'))
);

CREATE TABLE PACKAGE (

CREATE TABLE PROMOTION (
  PromotionID CHAR(6) NOT NULL,
  PromoName VARCHAR2(100) NOT NULL,
  DiscountPercentage NUMBER(5,2),
  MinSpentAmount NUMBER(6,2) DEFAULT 0,
  MaxDiscountAmount NUMBER(6,2),
  MaxRedemptions NUMBER(6) NOT NULL,
  CurrentRedemptions NUMBER(6) DEFAULT 0 NOT NULL,
  StartDate DATE NOT NULL,
  EndDate DATE NOT NULL,
  CONSTRAINT pk_promo_promotionid PRIMARY KEY (PromotionID),
  CONSTRAINT chk_promo_promotionid CHECK (REGEXP_LIKE(PromotionID, '^P[0-9]{5}$')),
  CONSTRAINT chk_promo_discountpercentage CHECK (DiscountPercentage >= 0 AND DiscountPercentage <= 100),
  CONSTRAINT chk_promo_minspentamount CHECK (MinSpentAmount >= 0),
  CONSTRAINT chk_promo_maxdiscountamount CHECK (MaxDiscountAmount >= 0),
  CONSTRAINT chk_promo_maxredemptions CHECK (MaxRedemptions > 0),
  CONSTRAINT chk_promo_currentredemptions CHECK (CurrentRedemptions >= 0),
  CONSTRAINT chk_promo_enddate CHECK (EndDate >= StartDate)
);

CREATE TABLE SUBSCRIPTION (

  EndDate DATE NOT NULL,
  Status VARCHAR2(20) DEFAULT 'Active' NOT NULL,
  CONSTRAINT pk_sub_subscriptionid PRIMARY KEY (SubscriptionID),
  CONSTRAINT fk_sub_memberid FOREIGN KEY (MemberID) REFERENCES MEMBER(MemberID),
  CONSTRAINT fk_sub_subscriptionplanid FOREIGN KEY (SubscriptionPlanID) REFERENCES SUBSCRIPTIONPLAN(SubscriptionPlanID),

CREATE TABLE "ORDER" (
  OrderID CHAR(6) NOT NULL,
  MemberID CHAR(6) NOT NULL,
  DeliveryID CHAR(6),
  MemberVoucherID CHAR(7),
  PickupType VARCHAR2(50) DEFAULT 'Delivery' NOT NULL,
  OrderDate DATE DEFAULT SYSDATE NOT NULL,
  OrderTime VARCHAR2(20) NOT NULL,
  TotalAmount NUMBER(8,2) DEFAULT 0.00 NOT NULL,
  OrderStatus VARCHAR2(50) DEFAULT 'Pending' NOT NULL,
  CONSTRAINT pk_ord_orderid PRIMARY KEY (OrderID),
  CONSTRAINT fk_ord_memberid FOREIGN KEY (MemberID) REFERENCES MEMBER(MemberID),
  CONSTRAINT fk_ord_deliveryid FOREIGN KEY (DeliveryID) REFERENCES DELIVERY(DeliveryID),
  CONSTRAINT fk_ord_membervoucherid FOREIGN KEY (MemberVoucherID) REFERENCES MEMBERVOUCHER(MemberVoucherID),
  CONSTRAINT chk_ord_orderid CHECK (REGEXP_LIKE(OrderID, '^O[0-9]{5}$')),
  CONSTRAINT chk_ord_pickuptype CHECK (PickupType IN ('Delivery', 'Pickup')),
  CONSTRAINT chk_ord_totalamount CHECK (TotalAmount >= 0),
  CONSTRAINT chk_ord_orderstatus CHECK (OrderStatus IN ('Pending', 'Preparing', 'Delivering', 'Completed', 'Cancelled'))
);
