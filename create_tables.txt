CREATE TABLE ADDRESS (
  AddressID CHAR(6) NOT NULL,
  AddressLine VARCHAR2(255) NOT NULL,
  City VARCHAR2(100) NOT NULL,
  State VARCHAR2(100) NOT NULL,
  Postcode CHAR(5) NOT NULL,
  CONSTRAINT pk_a_addressid PRIMARY KEY (AddressID),
  CONSTRAINT chk_a_addressid CHECK (REGEXP_LIKE(AddressID, '^A[0-9]{5}$')),
  CONSTRAINT chk_a_postcode CHECK (REGEXP_LIKE(Postcode, '^[0-9]{5}$'))
);

CREATE TABLE CATEGORY (
  CategoryID CHAR(6) NOT NULL,
  CategoryName VARCHAR2(100) NOT NULL,
  Description VARCHAR2(255),
  CONSTRAINT pk_c_categoryid PRIMARY KEY (CategoryID),
  CONSTRAINT chk_c_categoryid CHECK (REGEXP_LIKE(CategoryID, '^C[0-9]{5}$')),
  CONSTRAINT uq_c_categoryname UNIQUE (CategoryName)
);

CREATE TABLE VOUCHERTYPE (
  VoucherTypeID CHAR(7) NOT NULL,
  TypeName VARCHAR2(100) NOT NULL,
  Description VARCHAR2(255),
  CONSTRAINT pk_vt_vouchertypeid PRIMARY KEY (VoucherTypeID),
  CONSTRAINT chk_vt_vouchertypeid CHECK (REGEXP_LIKE(VoucherTypeID, '^VT[0-9]{5}$')),
  CONSTRAINT uq_vt_typename UNIQUE (TypeName)
);

CREATE TABLE SUBSCRIPTIONPLAN (
  SubscriptionPlanID CHAR(7) NOT NULL,
  PlanName VARCHAR2(100) NOT NULL,
  DurationInDays NUMBER(4) NOT NULL,
  Price NUMBER(6,2) DEFAULT 0.00 NOT NULL,
  CONSTRAINT pk_sp_subscriptionplanid PRIMARY KEY (SubscriptionPlanID),
  CONSTRAINT chk_sp_subscriptionplanid CHECK (REGEXP_LIKE(SubscriptionPlanID, '^SP[0-9]{5}$')),
  CONSTRAINT chk_sp_durationindays CHECK (DurationInDays > 0),
  CONSTRAINT chk_sp_price CHECK (Price >= 0)
);

CREATE TABLE "SETS" (
  SetID CHAR(7) NOT NULL,
  SetName VARCHAR2(100) NOT NULL,
  Discount NUMBER(5,2) DEFAULT 0,
  CONSTRAINT pk_se_setid PRIMARY KEY (SetID),
  CONSTRAINT chk_se_setid CHECK (REGEXP_LIKE(SetID, '^ST[0-9]{5}$')),
  CONSTRAINT chk_se_discount CHECK (Discount >= 0 AND Discount <= 100)

CREATE TABLE PACKAGE (
  PackageID CHAR(7) NOT NULL,
  PackageName VARCHAR2(100) NOT NULL,
  DiscountedPrice NUMBER(6,2) NOT NULL,
  CONSTRAINT pk_pa_packageid PRIMARY KEY (PackageID),
  CONSTRAINT chk_pa_packageid CHECK (REGEXP_LIKE(PackageID, '^PK[0-9]{5}$')),
  CONSTRAINT chk_pa_discountedprice CHECK (DiscountedPrice >= 0)
);

CREATE TABLE PROMOTION (
  PromotionID CHAR(6) NOT NULL,
  PromoName VARCHAR2(100) NOT NULL,
  DiscountPercentage NUMBER(5,2),
  MinSpentAmount NUMBER(6,2) DEFAULT 0,
  MaxDiscountAmount NUMBER(6,2),
  MaxRedemptions NUMBER(6) NOT NULL,
  CurrentRedemptions NUMBER(6) DEFAULT 0 NOT NULL,
  StartDate DATE NOT NULL,
  EndDate DATE NOT NULL,
  CONSTRAINT pk_pr_promotionid PRIMARY KEY (PromotionID),
  CONSTRAINT chk_pr_promotionid CHECK (REGEXP_LIKE(PromotionID, '^P[0-9]{5}$')),
  CONSTRAINT chk_pr_discountpercentage CHECK (DiscountPercentage >= 0 AND DiscountPercentage <= 100),
  CONSTRAINT chk_pr_minspentamount CHECK (MinSpentAmount >= 0),
  CONSTRAINT chk_pr_maxdiscountamount CHECK (MaxDiscountAmount >= 0),
  CONSTRAINT chk_pr_maxredemptions CHECK (MaxRedemptions > 0),
  CONSTRAINT chk_pr_currentredemptions CHECK (CurrentRedemptions >= 0),
  CONSTRAINT chk_pr_enddate CHECK (EndDate >= StartDate)
);

CREATE TABLE MEMBER (
  MemberID CHAR(6) NOT NULL,
  FullName VARCHAR2(150) NOT NULL,
  Email VARCHAR2(100) NOT NULL,
  PhoneNo VARCHAR2(20) NOT NULL,
  Password VARCHAR2(255) NOT NULL,
  RegistrationDate DATE DEFAULT SYSDATE NOT NULL,
  CONSTRAINT pk_m_memberid PRIMARY KEY (MemberID),
  CONSTRAINT chk_m_memberid CHECK (REGEXP_LIKE(MemberID, '^M[0-9]{5}$')),
  CONSTRAINT uq_m_email UNIQUE (Email)

CREATE TABLE RESTAURANT (
  RestaurantID CHAR(6) NOT NULL,
  AddressID CHAR(6) NOT NULL,
  RestaurantName VARCHAR2(150) NOT NULL,
  ContactNo VARCHAR2(20) NOT NULL,
  AverageRating NUMBER(3,2) DEFAULT 0.00,
  IsHalal CHAR(1) NOT NULL,
  OperationHours VARCHAR2(255),
  CONSTRAINT pk_r_restaurantid PRIMARY KEY (RestaurantID),
  CONSTRAINT fk_r_addressid FOREIGN KEY (AddressID) REFERENCES ADDRESS(AddressID),
  CONSTRAINT chk_r_restaurantid CHECK (REGEXP_LIKE(RestaurantID, '^R[0-9]{5}$')),
  CONSTRAINT chk_r_averagerating CHECK (AverageRating >= 1 AND AverageRating <= 5),
  CONSTRAINT chk_r_ishalal CHECK (IsHalal IN ('Y', 'N'))
);

CREATE TABLE DELIVERYRIDER (
  DeliveryRiderID CHAR(7) NOT NULL,

  CONSTRAINT chk_dr_deliveryriderid CHECK (REGEXP_LIKE(DeliveryRiderID, '^DR[0-9]{5}$'))
);

CREATE TABLE VOUCHER (

CREATE TABLE MEMBERADDRESS (
  MemberID CHAR(6) NOT NULL,
  AddressID CHAR(6) NOT NULL,
  CONSTRAINT pk_ma_memberid_addressid PRIMARY KEY (MemberID, AddressID),
  CONSTRAINT fk_ma_memberid FOREIGN KEY (MemberID) REFERENCES MEMBER(MemberID),
  CONSTRAINT fk_ma_addressid FOREIGN KEY (AddressID) REFERENCES ADDRESS(AddressID)

CREATE TABLE ITEM (
  ItemID CHAR(6) NOT NULL,
  RestaurantID CHAR(6) NOT NULL,
  CategoryID CHAR(6) NOT NULL,
  ItemName VARCHAR2(150) NOT NULL,
  Description VARCHAR2(255),
  Price NUMBER(6,2) NOT NULL,
  IsAvailable CHAR(1) DEFAULT 'Y' NOT NULL,
  CONSTRAINT pk_i_itemid PRIMARY KEY (ItemID),
  CONSTRAINT fk_i_restaurantid FOREIGN KEY (RestaurantID) REFERENCES RESTAURANT(RestaurantID),
  CONSTRAINT fk_i_categoryid FOREIGN KEY (CategoryID) REFERENCES CATEGORY(CategoryID),
  CONSTRAINT chk_i_itemid CHECK (REGEXP_LIKE(ItemID, '^I[0-9]{5}$')),
  CONSTRAINT chk_i_price CHECK (Price >= 0),
  CONSTRAINT chk_i_isavailable CHECK (IsAvailable IN ('Y', 'N'))
);

CREATE TABLE SUBSCRIPTION (
  SubscriptionID CHAR(6) NOT NULL,
  MemberID CHAR(6) NOT NULL,
  SubscriptionPlanID CHAR(7) NOT NULL,
  StartDate DATE NOT NULL,
  EndDate DATE NOT NULL,
  Status VARCHAR2(20) DEFAULT 'Active' NOT NULL,
  CONSTRAINT pk_su_subscriptionid PRIMARY KEY (SubscriptionID),
  CONSTRAINT fk_su_memberid FOREIGN KEY (MemberID) REFERENCES MEMBER(MemberID),
  CONSTRAINT fk_su_subscriptionplanid FOREIGN KEY (SubscriptionPlanID) REFERENCES SUBSCRIPTIONPLAN(SubscriptionPlanID),
  CONSTRAINT chk_su_subscriptionid CHECK (REGEXP_LIKE(SubscriptionID, '^S[0-9]{5}$')),
  CONSTRAINT chk_su_enddate CHECK (EndDate > StartDate),
  CONSTRAINT chk_su_status CHECK (Status IN ('Active', 'Expired', 'Cancelled'))
);

CREATE TABLE DELIVERY (
  DeliveryID CHAR(6) NOT NULL,

  AddressID CHAR(6) NOT NULL,

  CONSTRAINT fk_d_deliveryriderid FOREIGN KEY (DeliveryRiderID) REFERENCES DELIVERYRIDER(DeliveryRiderID),
  CONSTRAINT fk_d_addressid FOREIGN KEY (AddressID) REFERENCES ADDRESS(AddressID),
  CONSTRAINT chk_d_deliveryid CHECK (REGEXP_LIKE(DeliveryID, '^D[0-9]{5}$')),
  CONSTRAINT chk_d_distance CHECK (Distance > 0),
  CONSTRAINT chk_d_deliveryprice CHECK (DeliveryPrice >= 0)
);

CREATE TABLE ITEMPROMOTION (

CREATE TABLE "ORDERS" (
  OrderID CHAR(6) NOT NULL,
  MemberID CHAR(6) NOT NULL,
  DeliveryID CHAR(6),
  MemberVoucherID CHAR(7),
  PickupType VARCHAR2(50) DEFAULT 'Delivery' NOT NULL,
  OrderDate DATE DEFAULT SYSDATE NOT NULL,
  OrderTime VARCHAR2(20) NOT NULL,
  TotalAmount NUMBER(8,2) DEFAULT 0.00 NOT NULL,
  OrderStatus VARCHAR2(50) DEFAULT 'Pending' NOT NULL,
  CONSTRAINT pk_o_orderid PRIMARY KEY (OrderID),
  CONSTRAINT fk_o_memberid FOREIGN KEY (MemberID) REFERENCES MEMBER(MemberID),
  CONSTRAINT fk_o_deliveryid FOREIGN KEY (DeliveryID) REFERENCES DELIVERY(DeliveryID),
  CONSTRAINT fk_o_membervoucherid FOREIGN KEY (MemberVoucherID) REFERENCES MEMBERVOUCHER(MemberVoucherID),
  CONSTRAINT chk_o_orderid CHECK (REGEXP_LIKE(OrderID, '^O[0-9]{5}$')),
  CONSTRAINT chk_o_pickuptype CHECK (PickupType IN ('Delivery', 'Pickup')),
  CONSTRAINT chk_o_totalamount CHECK (TotalAmount >= 0),
  CONSTRAINT chk_o_orderstatus CHECK (OrderStatus IN ('Pending', 'Preparing', 'Delivering', 'Completed', 'Cancelled'))
);

CREATE TABLE DELIVERYFEEDBACK (

CREATE TABLE PAYMENT (
  PaymentID CHAR(7) NOT NULL,
  OrderID CHAR(6),
  SubscriptionID CHAR(6),

  PaymentStatus VARCHAR2(50) DEFAULT 'Pending' NOT NULL,

  CONSTRAINT chk_pm_paymentstatus CHECK (PaymentStatus IN ('Pending', 'Completed', 'Failed'))
);
