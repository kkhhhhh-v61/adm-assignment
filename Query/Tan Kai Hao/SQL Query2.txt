CL SCR

SET SERVEROUTPUT ON FORMAT WRAPPED
SET FEEDBACK OFF
SET LINESIZE 88
SET PAGESIZE 100

-- View to combine Orders with 'Cancelled' status from Package, Set, and Item
CREATE OR REPLACE VIEW vw_cancelled_revenue AS
SELECT 
    O.OrderID,
    R.RestaurantName,
    (OI.Quantity * OI.PriceEach) AS LostRevenue
FROM ORDERS O
JOIN ORDERITEM OI ON O.OrderID = OI.OrderID
JOIN ITEM I ON OI.ItemID = I.ItemID
JOIN RESTAURANT R ON I.RestaurantID = R.RestaurantID
WHERE O.OrderStatus = 'Cancelled'
UNION ALL
SELECT 
    O.OrderID,
    R.RestaurantName,
    (OP.Quantity * OP.PriceEach) AS LostRevenue
FROM ORDERS O
JOIN ORDERPACKAGE OP ON O.OrderID = OP.OrderID
JOIN (
    SELECT DISTINCT PI.PackageID, I.RestaurantID 
    FROM PACKAGEITEM PI
    JOIN ITEM I ON PI.ItemID = I.ItemID
) PR ON OP.PackageID = PR.PackageID
JOIN RESTAURANT R ON PR.RestaurantID = R.RestaurantID
WHERE O.OrderStatus = 'Cancelled';

DECLARE
    -- Cursor to sum total cancelled orders and lost revenue by restaurant
    CURSOR c_cancel IS
        SELECT
            RestaurantName,
            COUNT(DISTINCT OrderID) AS CANCELLED_ORDERS,
            SUM(LostRevenue) AS LOST_REVENUE,
            -- calculate lost revenue share percentage
            ROUND(RATIO_TO_REPORT(SUM(LostRevenue)) OVER () * 100, 2) AS IMPACT_SHARE
        FROM vw_cancelled_revenue
        GROUP BY RestaurantName
        ORDER BY LOST_REVENUE DESC;
        
    -- variable declaration and initialization
    v_grand_orders  NUMBER := 0;
    v_grand_revenue NUMBER := 0;
    v_grand_share   NUMBER := 0;
    
    v_top_restaurant VARCHAR2(150);
    v_top_orders     NUMBER;
    v_top_revenue    NUMBER;
    v_top_impact     NUMBER;
    v_is_first       BOOLEAN := TRUE;
BEGIN
    -- print title and header
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 88, '='));
    DBMS_OUTPUT.PUT_LINE(LPAD(' ', 20, ' ') || 'LOST REVENUE FROM INCOMPLETE ORDERS BY RESTAURANT');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 88, '='));
    DBMS_OUTPUT.PUT_LINE(
        RPAD('RESTAURANT NAME', 36, ' ') || ' | ' ||
        LPAD('CANCELLED ORDERS', 16, ' ') || ' | ' ||
        LPAD('LOST REVENUE (RM)', 17, ' ') || ' | ' ||
        LPAD('IMPACT (%)', 10, ' ')
    );
    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 36, '-') || '-+-' ||
        RPAD('-', 16, '-') || '-+-' ||
        RPAD('-', 17, '-') || '-+-' ||
        RPAD('-', 10, '-')
    );

    FOR r IN c_cancel LOOP
        -- assign first row of data as top
        IF v_is_first THEN
            v_top_restaurant := r.RestaurantName;
            v_top_orders     := r.CANCELLED_ORDERS;
            v_top_revenue    := r.LOST_REVENUE;
            v_top_impact     := r.IMPACT_SHARE;
            v_is_first       := FALSE;
        END IF;
        
        -- calculate grand total
        v_grand_orders  := v_grand_orders + r.CANCELLED_ORDERS;
        v_grand_revenue := v_grand_revenue + r.LOST_REVENUE;
        v_grand_share   := v_grand_share + r.IMPACT_SHARE;
        
        -- print column values
        DBMS_OUTPUT.PUT_LINE(
            RPAD(r.RestaurantName, 36, ' ') || ' | ' ||
            LPAD(TO_CHAR(r.CANCELLED_ORDERS, '999,990'), 16, ' ') || ' | ' ||
            LPAD(TO_CHAR(r.LOST_REVENUE, '$99,999,990.99'), 17, ' ') || ' | ' ||
            LPAD(TO_CHAR(r.IMPACT_SHARE, '990.00'), 10, ' ')
        );
    END LOOP;

    -- print separator
    DBMS_OUTPUT.PUT_LINE(
        RPAD('-', 36, '-') || '-+-' ||
        RPAD('-', 16, '-') || '-+-' ||
        RPAD('-', 17, '-') || '-+-' ||
        RPAD('-', 10, '-')
    );
    
    -- print grand total values
    DBMS_OUTPUT.PUT_LINE(
        RPAD('GRAND TOTAL :', 36, ' ') || ' | ' ||
        LPAD(TO_CHAR(v_grand_orders, '999,990'), 16, ' ') || ' | ' ||
        LPAD(TO_CHAR(v_grand_revenue, '$99,999,990.99'), 17, ' ') || ' | ' ||
        LPAD(TO_CHAR(v_grand_share, '990.00'), 10, ' ')
    );
    
    -- print summary
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 88, '='));
    DBMS_OUTPUT.PUT_LINE(' [ LOST REVENUE SUMMARY ]');
    DBMS_OUTPUT.PUT_LINE(' >> MOST AFFECTED RESTAURANT : ' || v_top_restaurant);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL CANCELLED ORDERS   : ' || v_top_orders);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL POTENTIAL LOSS     : RM ' || TO_CHAR(v_top_revenue, 'FM99,999,990.99'));
    DBMS_OUTPUT.PUT_LINE(' >> OVERALL IMPACT SHARE     : ' || v_top_impact || '%');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 88, '='));
END;
