COLUMN DeliveryID FORMAT A10
COLUMN DeliveryRiderID FORMAT A15
COLUMN OrderStatus FORMAT A11
COLUMN MAX(DeliveryID) FORMAT A15

SELECT MAX(DELIVERYID) FROM DELIVERY;

SELECT OrderID, DeliveryID, OrderStatus
FROM ORDERS
WHERE OrderID = 'O10005';
