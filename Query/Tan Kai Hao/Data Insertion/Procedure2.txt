PROMPT CASE 5: ITEM NOT FOUND / UNAVAILABLE

COLUMN MAX(PackageID) FORMAT A14
SELECT MAX(PackageID) FROM PACKAGE;

PROMPT
PROMPT ====================================================================================================
PROMPT CASE 6: VALID PACKAGE CREATION (2 ITEMS)
PROMPT Description: Package with 2 items from the same restaurant with a 15% discount.
EXEC prc_create_package('I10001', 2, 'I10002', 1, 0.15);

COLUMN PackageID FORMAT A10
COLUMN ItemID FORMAT A7
COLUMN Quantity FORMAT 9999
COLUMN OriginalPrice FORMAT 9999.99
COLUMN Discount FORMAT 0.99

SELECT P.PackageID, PI.ItemID, PI.Quantity, P.OriginalPrice, P.Discount
FROM PACKAGE P
JOIN PACKAGEITEM PI ON P.PackageID = PI.PackageID
WHERE P.PackageID = (SELECT MAX(PackageID) FROM PACKAGE);
