CL SCR

-- Create view for member spending analysis
CREATE OR REPLACE VIEW VIEW_MEMBER_SPENDING AS
SELECT
    M.MemberID,
    M.FullName,
    COUNT(O.OrderID) AS TOTAL_ORDERS,
    SUM(O.TotalAmount) AS TOTAL_SPENDING
FROM MEMBER M
JOIN ORDERS O ON M.MemberID = O.MemberID
WHERE O.OrderStatus = 'Completed'
GROUP BY M.MemberID, M.FullName;

-- SQL*Plus settings
SET LINESIZE 170
SET PAGESIZE 200
SET FEEDBACK OFF
SET SQLBLANKLINES ON
SET SERVEROUTPUT ON

-- Format columns
COLUMN MEMBERID FORMAT A13 HEADING 'MEMBER ID'
COLUMN FULLNAME FORMAT A25 HEADING 'MEMBER NAME'
COLUMN TOTAL_ORDERS FORMAT 999,990 HEADING 'TOTAL ORDERS'
COLUMN ORDER_SHARE FORMAT 990.00 HEADING 'ORDER SHARE (%)'
COLUMN TOTAL_SPENDING FORMAT 999,990.99 HEADING 'TOTAL SPENDING (RM)'
COLUMN SPENDING_SHARE FORMAT 990.00 HEADING 'SPENDING SHARE (%)'
COLUMN LOYALTY_TIER FORMAT A12 HEADING 'LOYALTY TIER' 

-- Report header
TTITLE LEFT '========================================================================================================================' SKIP 1 -
       LEFT '                                 MEMBER ORDER FREQUENCY AND SPENDING ANALYSIS REPORT                                   ' SKIP 1 -
       LEFT '========================================================================================================================' SKIP 2

-- Totals
BREAK ON REPORT
COMPUTE SUM LABEL 'GRAND TOTAL :' OF TOTAL_ORDERS TOTAL_SPENDING ON REPORT

-- Main query
SELECT
    MemberID,
    FullName,
    TOTAL_ORDERS,
    ROUND(RATIO_TO_REPORT(TOTAL_ORDERS) OVER () * 100, 2) AS ORDER_SHARE,
    TOTAL_SPENDING,
    ROUND(RATIO_TO_REPORT(TOTAL_SPENDING) OVER () * 100, 2) AS SPENDING_SHARE,

    LPAD(
    CASE
        WHEN TOTAL_SPENDING >= 800 THEN 'Platinum'
        WHEN TOTAL_SPENDING >= 500 THEN 'Gold'
        ELSE 'Silver'
    END, 12
	) AS LOYALTY_TIER

FROM VIEW_MEMBER_SPENDING
ORDER BY TOTAL_SPENDING DESC;

-- Disable header for summary
TTITLE OFF
CLEAR BREAKS
CLEAR COMPUTES
SET HEADING OFF

-- PL/SQL Executive Summary
DECLARE
    v_member VARCHAR2(100);
    v_memberid VARCHAR2(10);
    v_orders NUMBER;
    v_spending NUMBER;
    v_loyalty VARCHAR2(12);
BEGIN

    SELECT MemberID,
           FullName,
           TOTAL_ORDERS,
           TOTAL_SPENDING,
           CASE
               WHEN TOTAL_SPENDING >= 800 THEN 'Platinum'
               WHEN TOTAL_SPENDING >= 500 THEN 'Gold'
               ELSE 'Silver'
           END
    INTO v_memberid, v_member, v_orders, v_spending, v_loyalty
    FROM (
        SELECT MemberID, FullName, TOTAL_ORDERS, TOTAL_SPENDING
        FROM VIEW_MEMBER_SPENDING
        ORDER BY TOTAL_SPENDING DESC
    )
    WHERE ROWNUM = 1;

    DBMS_OUTPUT.PUT_LINE('------------------------------------------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(' [ MEMBER SPENDING SUMMARY ]');
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE(' >> TOP MEMBER ID      : ' || v_memberid);
    DBMS_OUTPUT.PUT_LINE(' >> MEMBER NAME        : ' || v_member);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL ORDERS       : ' || v_orders);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL SPENDING     : RM ' || v_spending);
    DBMS_OUTPUT.PUT_LINE(' >> LOYALTY TIER       : ' || v_loyalty);
    DBMS_OUTPUT.PUT_LINE('------------------------------------------------------------------------------------------------------------------------');

END;
/

-- Reset settings
SET HEADING ON
SET FEEDBACK ON
CLEAR COLUMNS
