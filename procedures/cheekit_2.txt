CREATE OR REPLACE PROCEDURE PR_PROCESS_DELIVERY_HANDOVER (
    p_order_id IN ORDERS.OrderID%TYPE
) AS
    v_order_rec      ORDERS%ROWTYPE;
    v_delivery_rec   DELIVERY%ROWTYPE;
    
    v_order_count    NUMBER;
    v_time_diff_mins NUMBER;
    v_new_voucher_id VOUCHER.VoucherID%TYPE;
    v_new_mv_id      MEMBERVOUCHER.MemberVoucherID%TYPE;
    
    e_invalid_order  EXCEPTION;

BEGIN
    SAVEPOINT start_handover;

    -- 1. Validate if the Order actually exists
    SELECT COUNT(*) INTO v_order_count FROM ORDERS WHERE OrderID = p_order_id;
    IF v_order_count = 0 THEN
        RAISE e_invalid_order;
    END IF;

    -- 2. Lock tables to prevent concurrency conflicts with triggers
    LOCK TABLE ORDERS, VOUCHER, MEMBERVOUCHER IN EXCLUSIVE MODE;

    -- 3. Fetch the Order record
    SELECT * INTO v_order_rec FROM ORDERS WHERE OrderID = p_order_id;

    DBMS_OUTPUT.PUT_LINE('==================================================');
    DBMS_OUTPUT.PUT_LINE('[DELIVERY STATE MACHINE] Target Order: ' || p_order_id);
    DBMS_OUTPUT.PUT_LINE('Current Order Status : ' || v_order_rec.OrderStatus);
    DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');

    -- ==========================================
    -- STATE 1: PENDING -> DELIVERING
    -- ==========================================
    IF v_order_rec.OrderStatus = 'Pending' THEN
        
        UPDATE ORDERS SET OrderStatus = 'Delivering' WHERE OrderID = p_order_id;

        DBMS_OUTPUT.PUT_LINE('Action  : Food Picked Up by Rider.');
        DBMS_OUTPUT.PUT_LINE('Result  : Order Status changed to "Delivering".');

    -- ==========================================
    -- STATE 2: DELIVERING -> COMPLETED (With Apology Check)
    -- ==========================================
    ELSIF v_order_rec.OrderStatus = 'Delivering' THEN
        
        UPDATE ORDERS SET OrderStatus = 'Completed' WHERE OrderID = p_order_id;

        DBMS_OUTPUT.PUT_LINE('Action  : Food Delivered to Customer.');
        DBMS_OUTPUT.PUT_LINE('Result  : Order Status changed to "Completed".');

        -- Check if this order actually has a Delivery attached (not a Pickup)
        IF v_order_rec.DeliveryID IS NOT NULL THEN
            SELECT * INTO v_delivery_rec FROM DELIVERY WHERE DeliveryID = v_order_rec.DeliveryID;

            BEGIN
                -- Calculate wait time using the exact MOD logic from your analytical query
                v_time_diff_mins := MOD((TO_DATE(v_delivery_rec.DeliveryTime, 'HH24:MI') - TO_DATE(v_order_rec.OrderTime, 'HH24:MI')) * 24 * 60 + 1440, 1440);
                DBMS_OUTPUT.PUT_LINE('Metrics : Wait Time = ' || v_time_diff_mins || ' mins.');

                -- Auto-Apology Logic
                IF v_time_diff_mins > 60 THEN
                    -- Generate new VoucherID safely
                    SELECT 'V' || TO_CHAR(NVL(MAX(TO_NUMBER(SUBSTR(VoucherID, 2))), 0) + 1, 'FM00000') 
                    INTO v_new_voucher_id FROM VOUCHER;

                    -- Insert the new apology voucher
                    INSERT INTO VOUCHER (VoucherID, VoucherTypeID, VoucherName, VoucherCode, DiscountPercentage, DiscountAmount, MinSpendAmount, MaxDiscountAmount, ExpiryDate)
                    VALUES (v_new_voucher_id, 'VT00001', 'Apology Voucher - Delay', 'SRY' || v_new_voucher_id, NULL, 10.00, 0, 10.00, SYSDATE + 30);

                    -- Generate new MemberVoucherID
                    SELECT 'MV' || TO_CHAR(NVL(MAX(TO_NUMBER(SUBSTR(MemberVoucherID, 3))), 0) + 1, 'FM00000') 
                    INTO v_new_mv_id FROM MEMBERVOUCHER;

                    -- Link the apology voucher to the Member who suffered the delay
                    INSERT INTO MEMBERVOUCHER (MemberVoucherID, MemberID, VoucherID, DateClaimed, IsUsed)
                    VALUES (v_new_mv_id, v_order_rec.MemberID, v_new_voucher_id, SYSDATE, 'N');

                    DBMS_OUTPUT.PUT_LINE('🚨 ALERT : SEVERE BOTTLENECK DETECTED (> 60 MINS)!');
                    DBMS_OUTPUT.PUT_LINE('Action  : Auto-Apology Triggered.');
                    DBMS_OUTPUT.PUT_LINE('Result  : Voucher ' || v_new_voucher_id || ' deposited to Member ' || v_order_rec.MemberID);
                ELSE
                    DBMS_OUTPUT.PUT_LINE('✅ SUCCESS: Delivery completed within acceptable timeframe.');
                END IF;
            EXCEPTION
                WHEN OTHERS THEN
                    DBMS_OUTPUT.PUT_LINE('Note    : Could not calculate time (String format mismatch).');
            END;
        END IF;

    -- ==========================================
    -- STATE 3: ALREADY COMPLETED OR CANCELLED
    -- ==========================================
    ELSIF v_order_rec.OrderStatus IN ('Completed', 'Cancelled') THEN
        DBMS_OUTPUT.PUT_LINE('Action  : Blocked. Order is already ' || v_order_rec.OrderStatus || '.');
    END IF;

    DBMS_OUTPUT.PUT_LINE('==================================================');
    COMMIT;

EXCEPTION
    WHEN e_invalid_order THEN
        ROLLBACK TO start_handover;
        DBMS_OUTPUT.PUT_LINE('ERROR: Order ID (' || p_order_id || ') does not exist in the database.');
    WHEN OTHERS THEN
        ROLLBACK TO start_handover;
        DBMS_OUTPUT.PUT_LINE('CRITICAL ERROR: ' || SQLERRM);
END;
