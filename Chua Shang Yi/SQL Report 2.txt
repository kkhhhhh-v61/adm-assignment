CREATE OR REPLACE PROCEDURE generate_voucher_report(
    p_start_date IN DATE,
    p_end_date   IN DATE
)
AS
    ------------------------------------------------------------
    -- NESTED CURSOR: Parent (Member) + Child (Aggregates)
    ------------------------------------------------------------
    CURSOR c_member_voucher IS
        SELECT 
            M.MemberID, 
            M.FullName,
            CURSOR(
                SELECT 
                    COUNT(DISTINCT O.OrderID) AS total_orders,
                    COUNT(O.MemberVoucherID) AS vouchers_used,
                    SUM(O.TotalAmount) AS total_spending,
                    SUM(
                        CASE 
                            WHEN V.DiscountAmount IS NOT NULL THEN V.DiscountAmount
                            ELSE NVL(O.TotalAmount * V.DiscountPercentage, 0)
                        END
                    ) AS total_discount
                FROM ORDERS O
                LEFT JOIN MEMBERVOUCHER MV ON O.MemberVoucherID = MV.MemberVoucherID
                LEFT JOIN VOUCHER V ON MV.VoucherID = V.VoucherID
                WHERE O.MemberID = M.MemberID
                  AND O.OrderStatus = 'Completed'
                  AND O.MemberVoucherID IS NOT NULL
                  AND TRUNC(O.OrderDate) BETWEEN p_start_date AND p_end_date
            ) AS stats_cursor
        FROM MEMBER M
        WHERE EXISTS (
            SELECT 1 FROM ORDERS O 
            WHERE O.MemberID = M.MemberID 
              AND O.OrderStatus = 'Completed'
              AND O.MemberVoucherID IS NOT NULL
              AND TRUNC(O.OrderDate) BETWEEN p_start_date AND p_end_date
        );

    ------------------------------------------------------------
    -- VOUCHER TYPE PERFORMANCE (Kept Original)
    ------------------------------------------------------------
    CURSOR c_voucher_type IS
        SELECT
            VT.TypeName,
            COUNT(O.OrderID) total_orders,
            SUM(O.TotalAmount) revenue,
            SUM(
                CASE
                    WHEN V.DiscountAmount IS NOT NULL THEN V.DiscountAmount
                    ELSE O.TotalAmount * V.DiscountPercentage
                END
            ) total_discount
        FROM ORDERS O
        JOIN MEMBERVOUCHER MV ON O.MemberVoucherID = MV.MemberVoucherID
        JOIN VOUCHER V ON MV.VoucherID = V.VoucherID
        JOIN VOUCHERTYPE VT ON V.VoucherTypeID = VT.VoucherTypeID
        WHERE O.OrderStatus='Completed'
          AND TRUNC(O.OrderDate) BETWEEN p_start_date AND p_end_date
        GROUP BY VT.TypeName
        ORDER BY revenue DESC;

    -- Variables for Nested Cursor processing
    v_nested_cv      SYS_REFCURSOR;
    v_curr_mem_id    MEMBER.MemberID%TYPE;
    v_curr_mem_name  MEMBER.FullName%TYPE;
    v_curr_orders    NUMBER;
    v_curr_vouchers  NUMBER;
    v_curr_spending  NUMBER;
    v_curr_discount  NUMBER;

    -- Collection to store results for ranking and processing
    TYPE t_member_rec IS RECORD (
        FullName       VARCHAR2(150),
        total_orders   NUMBER,
        vouchers_used  NUMBER,
        total_spending NUMBER,
        total_discount NUMBER
    );
    TYPE t_member_tab IS TABLE OF t_member_rec INDEX BY PLS_INTEGER;
    v_mem_data t_member_tab;

    -- Original Global Variables
    v_total_orders   NUMBER := 0;
    v_total_vouchers NUMBER := 0;
    v_total_spending NUMBER := 0;
    v_total_discount NUMBER := 0;
    v_counter        NUMBER := 1;
    v_percentage     NUMBER;
    v_bar            VARCHAR2(100);
    v_color          VARCHAR2(20);
    v_top_member     VARCHAR2(120);
    v_top_spending   NUMBER := 0;
    v_has_data       BOOLEAN := FALSE;
    v_idx            NUMBER := 0;

    e_invalid_date EXCEPTION;
    e_no_data      EXCEPTION;

BEGIN
    -- DATE VALIDATION
    IF p_start_date > p_end_date THEN
        RAISE e_invalid_date;
    END IF;

    ------------------------------------------------------------
    -- FETCH DATA USING NESTED CURSOR
    ------------------------------------------------------------
    OPEN c_member_voucher;
    LOOP
        -- FIX: Fetch all 3 columns from the parent cursor
        FETCH c_member_voucher INTO v_curr_mem_id, v_curr_mem_name, v_nested_cv;
        EXIT WHEN c_member_voucher%NOTFOUND;

        -- Fetch the actual numbers from the child cursor
        FETCH v_nested_cv INTO v_curr_orders, v_curr_vouchers, v_curr_spending, v_curr_discount;
        CLOSE v_nested_cv;

        v_has_data := TRUE;
        v_idx := v_idx + 1;
        
        -- Store in PL/SQL table
        v_mem_data(v_idx).FullName       := v_curr_mem_name;
        v_mem_data(v_idx).total_orders   := v_curr_orders;
        v_mem_data(v_idx).vouchers_used  := v_curr_vouchers;
        v_mem_data(v_idx).total_spending := NVL(v_curr_spending, 0);
        v_mem_data(v_idx).total_discount := NVL(v_curr_discount, 0);

        -- Totals for ROI Analysis
        v_total_orders   := v_total_orders + v_curr_orders;
        v_total_vouchers := v_total_vouchers + v_curr_vouchers;
        v_total_spending := v_total_spending + NVL(v_curr_spending, 0);
        v_total_discount := v_total_discount + NVL(v_curr_discount, 0);

        -- Track Top Spender
        IF NVL(v_curr_spending, 0) > v_top_spending THEN
            v_top_spending := v_curr_spending;
            v_top_member   := v_curr_mem_name;
        END IF;
    END LOOP;
    CLOSE c_member_voucher;

    IF NOT v_has_data THEN
        RAISE e_no_data;
    END IF;

    -- Sort the PL/SQL table by spending (Descending) to recreate Ranking
    DECLARE
        v_temp t_member_rec;
    BEGIN
        FOR i IN 1..v_mem_data.COUNT-1 LOOP
            FOR j IN i+1..v_mem_data.COUNT LOOP
                IF v_mem_data(i).total_spending < v_mem_data(j).total_spending THEN
                    v_temp := v_mem_data(i);
                    v_mem_data(i) := v_mem_data(j);
                    v_mem_data(j) := v_temp;
                END IF;
            END LOOP;
        END LOOP;
    END;

    ------------------------------------------------------------
    -- REPORT HEADER (ORIGINAL)
    ------------------------------------------------------------
    DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');
    DBMS_OUTPUT.PUT_LINE('                                     VOUCHER REDEMPTION AND MEMBER SPENDING REPORT');
    DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');
    DBMS_OUTPUT.PUT_LINE('Period    : '||TO_CHAR(p_start_date,'DD-MON-YYYY')||' to '||TO_CHAR(p_end_date,'DD-MON-YYYY'));
    DBMS_OUTPUT.PUT_LINE('Generated : '||TO_CHAR(SYSDATE,'DD-MON-YYYY HH24:MI:SS'));
    DBMS_OUTPUT.PUT_LINE('-------------------------------------------------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(RPAD('RANK',6)||RPAD('MEMBER',25)||LPAD('ORDERS',10)||LPAD('VOUCHERS',12)||LPAD('SPENDING(RM)',15)||LPAD('DISCOUNT(RM)',15)||LPAD('NET(RM)',12)||LPAD('AVG(RM)',12)||'   SPENDING SHARE');
    DBMS_OUTPUT.PUT_LINE('-------------------------------------------------------------------------------------------------------------------------------');

    ------------------------------------------------------------
    -- MEMBER RESULTS LOOP (ORIGINAL FORMAT)
    ------------------------------------------------------------
    FOR i IN 1..v_mem_data.COUNT LOOP
        v_percentage := ROUND((v_mem_data(i).total_spending/v_total_spending)*100,2);

        IF v_percentage >= 20 THEN
            v_color := CHR(27)||'[31m'; -- RED
        ELSIF v_percentage >= 10 THEN
            v_color := CHR(27)||'[33m'; -- YELLOW
        ELSIF v_percentage >= 5 THEN
            v_color := CHR(27)||'[32m'; -- GREEN
        ELSE
            v_color := CHR(27)||'[36m'; -- CYAN
        END IF;

        v_bar := v_color || RPAD('#',ROUND(v_percentage/2),'#') || CHR(27)||'[0m';

        DBMS_OUTPUT.PUT_LINE(
            RPAD(i,6)||
            RPAD(v_mem_data(i).FullName,25)||
            LPAD(v_mem_data(i).total_orders,10)||
            LPAD(v_mem_data(i).vouchers_used,12)||
            LPAD(TO_CHAR(v_mem_data(i).total_spending,'999,999.99'),15)||
            LPAD(TO_CHAR(v_mem_data(i).total_discount,'999,999.99'),15)||
            LPAD(TO_CHAR(v_mem_data(i).total_spending - v_mem_data(i).total_discount,'999,999.99'),12)||
            LPAD(TO_CHAR(ROUND(v_mem_data(i).total_spending/v_mem_data(i).total_orders,2),'999,999.99'),12)||
            '   ' || v_bar
        );
    END LOOP;

    ------------------------------------------------------------
    -- EXECUTIVE SUMMARY (ORIGINAL)
    ------------------------------------------------------------
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('====================================================== EXECUTIVE SUMMARY ======================================================');
    DBMS_OUTPUT.PUT_LINE('Top Spending Member : '||v_top_member||' (RM '||TO_CHAR(v_top_spending,'999,999.99')||')');
    DBMS_OUTPUT.PUT_LINE('Total Voucher Used : '||v_total_vouchers);
    DBMS_OUTPUT.PUT_LINE('Total Discount Given : RM '||TO_CHAR(v_total_discount,'999,999.99'));
    DBMS_OUTPUT.PUT_LINE('Voucher Savings Rate : '||ROUND((v_total_discount/NULLIF(v_total_spending,0))*100,2)||'%');

    ------------------------------------------------------------
    -- VOUCHER TYPE PERFORMANCE (ORIGINAL)
    ------------------------------------------------------------
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('================================================== VOUCHER TYPE PERFORMANCE ===================================================');
    v_counter := 1; 
    FOR r IN c_voucher_type LOOP
        DBMS_OUTPUT.PUT_LINE(LPAD(v_counter || '.',4) || ' ' || RPAD(r.TypeName,30)||LPAD(r.total_orders,10)||' orders  '||LPAD('RM '||TO_CHAR(r.revenue,'999,999.99'),15)||' revenue');
        v_counter := v_counter + 1;
    END LOOP;

    ------------------------------------------------------------
    -- ROI ANALYSIS (ORIGINAL)
    ------------------------------------------------------------
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE('==================================================== VOUCHER ROI ANALYSIS =====================================================');
    DBMS_OUTPUT.PUT_LINE('Revenue Generated : RM '||TO_CHAR(v_total_spending,'999,999.99'));
    DBMS_OUTPUT.PUT_LINE('Total Discount Cost : RM '||TO_CHAR(v_total_discount,'999,999.99'));
    DBMS_OUTPUT.PUT_LINE('Voucher ROI : '||ROUND((v_total_spending-v_total_discount)/NULLIF(v_total_discount,0)*100,2)||'% return');
    DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');
    DBMS_OUTPUT.PUT_LINE('End of Report');

EXCEPTION
    WHEN e_invalid_date THEN
        RAISE_APPLICATION_ERROR(-20001,'Invalid date range.');
    WHEN e_no_data THEN
        DBMS_OUTPUT.PUT_LINE('No voucher redemption data found.');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);
END;
