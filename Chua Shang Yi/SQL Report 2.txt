CL SCR 
SET SERVEROUTPUT ON FORMAT WRAPPED;
SET LINESIZE 170
SET PAGESIZE 200

CREATE OR REPLACE PROCEDURE generate_voucher_report(
    p_start_date IN DATE,
    p_end_date   IN DATE
)
AS

------------------------------------------------------------
-- MEMBER VOUCHER ANALYSIS
------------------------------------------------------------

CURSOR c_member_voucher IS
SELECT
    DENSE_RANK() OVER (ORDER BY SUM(O.TotalAmount) DESC) AS spending_rank,
    M.MemberID,
    M.FullName,
    COUNT(DISTINCT O.OrderID) total_orders,
    COUNT(O.MemberVoucherID) vouchers_used,
    SUM(O.TotalAmount) total_spending,

    SUM(
        CASE
            WHEN V.DiscountAmount IS NOT NULL
            THEN V.DiscountAmount
            ELSE NVL(O.TotalAmount * V.DiscountPercentage,0)
        END
    ) total_discount

FROM MEMBER M
JOIN ORDERS O ON M.MemberID = O.MemberID
LEFT JOIN MEMBERVOUCHER MV ON O.MemberVoucherID = MV.MemberVoucherID
LEFT JOIN VOUCHER V ON MV.VoucherID = V.VoucherID

WHERE O.OrderStatus='Completed'
AND O.MemberVoucherID IS NOT NULL
AND TRUNC(O.OrderDate) BETWEEN p_start_date AND p_end_date

GROUP BY M.MemberID, M.FullName
ORDER BY total_spending DESC;

------------------------------------------------------------
-- VOUCHER TYPE PERFORMANCE
------------------------------------------------------------

CURSOR c_voucher_type IS
SELECT
    VT.TypeName,
    COUNT(O.OrderID) total_orders,
    SUM(O.TotalAmount) revenue,

    SUM(
        CASE
            WHEN V.DiscountAmount IS NOT NULL
            THEN V.DiscountAmount
            ELSE O.TotalAmount * V.DiscountPercentage
        END
    ) total_discount

FROM ORDERS O
JOIN MEMBERVOUCHER MV ON O.MemberVoucherID = MV.MemberVoucherID
JOIN VOUCHER V ON MV.VoucherID = V.VoucherID
JOIN VOUCHERTYPE VT ON V.VoucherTypeID = VT.VoucherTypeID

WHERE O.OrderStatus='Completed'
AND TRUNC(O.OrderDate) BETWEEN p_start_date AND p_end_date

GROUP BY VT.TypeName
ORDER BY revenue DESC;

------------------------------------------------------------
-- VARIABLES
------------------------------------------------------------

v_total_orders NUMBER:=0;
v_total_vouchers NUMBER:=0;
v_total_spending NUMBER:=0;
v_total_discount NUMBER:=0;
v_counter NUMBER := 1;
v_percentage NUMBER;
v_bar VARCHAR2(100);

v_color VARCHAR2(20);

v_top_member VARCHAR2(120);
v_top_spending NUMBER:=0;

e_invalid_date EXCEPTION;
e_no_data EXCEPTION;

v_has_data BOOLEAN:=FALSE;

BEGIN

------------------------------------------------------------
-- DATE VALIDATION
------------------------------------------------------------

IF p_start_date > p_end_date THEN
RAISE e_invalid_date;
END IF;

------------------------------------------------------------
-- CALCULATE TOTALS
------------------------------------------------------------

FOR r IN c_member_voucher LOOP

v_has_data := TRUE;

v_total_orders := v_total_orders + r.total_orders;
v_total_vouchers := v_total_vouchers + r.vouchers_used;
v_total_spending := v_total_spending + r.total_spending;
v_total_discount := v_total_discount + r.total_discount;

IF r.total_spending > v_top_spending THEN
v_top_spending := r.total_spending;
v_top_member := r.FullName;
END IF;

END LOOP;

IF NOT v_has_data THEN
RAISE e_no_data;
END IF;

------------------------------------------------------------
-- REPORT HEADER
------------------------------------------------------------

DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');
DBMS_OUTPUT.PUT_LINE('                                    VOUCHER REDEMPTION AND MEMBER SPENDING REPORT');
DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');

DBMS_OUTPUT.PUT_LINE('Period    : '||TO_CHAR(p_start_date,'DD-MON-YYYY')||' to '||TO_CHAR(p_end_date,'DD-MON-YYYY'));
DBMS_OUTPUT.PUT_LINE('Generated : '||TO_CHAR(SYSDATE,'DD-MON-YYYY HH24:MI:SS'));

DBMS_OUTPUT.PUT_LINE('-------------------------------------------------------------------------------------------------------------------------------');

DBMS_OUTPUT.PUT_LINE(
RPAD('RANK',6)||
RPAD('MEMBER',25)||
LPAD('ORDERS',10)||
LPAD('VOUCHERS',12)||
LPAD('SPENDING(RM)',15)||
LPAD('DISCOUNT(RM)',15)||
LPAD('NET(RM)',12)||
LPAD('AVG(RM)',12)||
'   SPENDING SHARE'
);

DBMS_OUTPUT.PUT_LINE('-------------------------------------------------------------------------------------------------------------------------------');

------------------------------------------------------------
-- MEMBER RESULTS (ONLY ENHANCED HERE)
------------------------------------------------------------

FOR r IN c_member_voucher LOOP

v_percentage := ROUND((r.total_spending/v_total_spending)*100,2);

/* ✅ COLOR LOGIC ADDED */
IF v_percentage >= 20 THEN
    v_color := CHR(27)||'[31m'; -- RED
ELSIF v_percentage >= 10 THEN
    v_color := CHR(27)||'[33m'; -- YELLOW
ELSIF v_percentage >= 5 THEN
    v_color := CHR(27)||'[32m'; -- GREEN
ELSE
    v_color := CHR(27)||'[36m'; -- CYAN
END IF;

/* ✅ KEEP '#' BUT ADD COLOR */
v_bar := v_color || RPAD('#',ROUND(v_percentage/2),'#') || CHR(27)||'[0m';

DBMS_OUTPUT.PUT_LINE(
RPAD(r.spending_rank,6)||
RPAD(r.FullName,25)||
LPAD(r.total_orders,10)||
LPAD(r.vouchers_used,12)||
LPAD(TO_CHAR(r.total_spending,'999,999.99'),15)||
LPAD(TO_CHAR(r.total_discount,'999,999.99'),15)||
LPAD(TO_CHAR(r.total_spending-r.total_discount,'999,999.99'),12)||
LPAD(TO_CHAR(ROUND(r.total_spending/r.total_orders,2),'999,999.99'),12)||
'   ' || v_bar
);

END LOOP;

------------------------------------------------------------
-- EXECUTIVE SUMMARY 
------------------------------------------------------------

DBMS_OUTPUT.PUT_LINE(' ');
DBMS_OUTPUT.PUT_LINE('====================================================== EXECUTIVE SUMMARY ======================================================');

DBMS_OUTPUT.PUT_LINE('Top Spending Member : '||v_top_member||
' (RM '||TO_CHAR(v_top_spending,'999,999.99')||')');

DBMS_OUTPUT.PUT_LINE('Total Voucher Used : '||v_total_vouchers);

DBMS_OUTPUT.PUT_LINE('Total Discount Given : RM '||TO_CHAR(v_total_discount,'999,999.99'));

DBMS_OUTPUT.PUT_LINE('Voucher Savings Rate : '||
ROUND((v_total_discount/v_total_spending)*100,2)||'%');

------------------------------------------------------------
-- VOUCHER TYPE PERFORMANCE
------------------------------------------------------------

DBMS_OUTPUT.PUT_LINE(' ');
DBMS_OUTPUT.PUT_LINE('================================================== VOUCHER TYPE PERFORMANCE ===================================================');

v_counter := 1; -- reset counter

FOR r IN c_voucher_type LOOP

DBMS_OUTPUT.PUT_LINE(
LPAD(v_counter || '.',4) || ' ' || 
RPAD(r.TypeName,30)||
LPAD(r.total_orders,10)||' orders  '||
LPAD('RM '||TO_CHAR(r.revenue,'999,999.99'),15)||
' revenue');

v_counter := v_counter + 1;

END LOOP;

------------------------------------------------------------
-- ROI ANALYSIS
------------------------------------------------------------

DBMS_OUTPUT.PUT_LINE(' ');
DBMS_OUTPUT.PUT_LINE('==================================================== VOUCHER ROI ANALYSIS =====================================================');

DBMS_OUTPUT.PUT_LINE('Revenue Generated : RM '||TO_CHAR(v_total_spending,'999,999.99'));
DBMS_OUTPUT.PUT_LINE('Total Discount Cost : RM '||TO_CHAR(v_total_discount,'999,999.99'));

DBMS_OUTPUT.PUT_LINE('Voucher ROI : '||
ROUND((v_total_spending-v_total_discount)/v_total_discount*100,2)||'% return');

DBMS_OUTPUT.PUT_LINE('===============================================================================================================================');
DBMS_OUTPUT.PUT_LINE('End of Report');

------------------------------------------------------------

EXCEPTION

WHEN e_invalid_date THEN
RAISE_APPLICATION_ERROR(-20001,'Invalid date range.');

WHEN e_no_data THEN
DBMS_OUTPUT.PUT_LINE('No voucher redemption data found.');

WHEN OTHERS THEN
DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);

END;
