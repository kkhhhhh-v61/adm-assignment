-- DECLARE A FUNCTION TO CALCULATE AGE AND RETURN AGE TO STORED PROCEDURE
CREATE OR REPLACE FUNCTION FUNC_GET_AGE(V_DOB IN DATE) RETURN NUMBER IS

BEGIN
	RETURN EXTRACT(YEAR FROM (SYSDATE - V_DOB) YEAR TO MONTH);
END;
/

	v_productname varchar2(70);
	v_oldprice number(7,2);
	v_newprice number(7,2);
BEGIN
-- GET THE PRODUCT PRICE
        SELECT ProductName, MSRP INTO v_productname, v_oldprice
           From PRODUCTS
           WHERE ProductCode = IN_Product_Code;

        v_newprice := round(v_oldprice *1.1,2);

-- UPDATE the record
        UPDATE PRODUCTS 
        SET MSRP = v_newprice
	      WHERE ProductCode = IN_Product_Code;

-- WRITE DATA TO THE PRICE_AUDIT TABLE   
        INSERT INTO Price_Audit VALUES(IN_Product_Code, v_productname, 
               v_oldprice, v_newprice, USER, SYSDATE);
	
        DBMS_OUTPUT.PUT_LINE('Product '||IN_Product_Code||' '||v_productname||
                ' new price is $' ||v_newprice);
END;

-- HEADING SECTION WITH INPUT PARAMETER
-- ACCEPT CUSTOMERNUMBER AS AN INPUT TO VALIDATE
CREATE OR REPLACE PROCEDURE PRC_LAST_ORDER (V_CUSTNO IN NUMBER) IS

-- DECLARATION SECTION
V_ORDERDATE DATE;
V_DURATION	NUMBER(4);   -- 9999 DAYS
V_MONTHS    NUMBER(4,1); -- 999.9 MONTHS 

-- EXECUTION SECTION
BEGIN
	-- SELECT INTO STATEMENT TO EXTRACT THE DATA FROM THE DATABASE
  -- EXTRACT LAST ORDERDATE FROM ORDER TABLE TO IDENTIFY LAST ORDER MADE
  SELECT MAX(ORDERDATE) INTO V_ORDERDATE 
  FROM   ORDERS
  WHERE  CUSTOMERNUMBER = V_CUSTNO;

	-- PRINT CUSTOMER INFORMATION AND ORDER INFORMATION
  IF (V_ORDERDATE IS NOT NULL) THEN
		DBMS_OUTPUT.PUT_LINE('CUSTOMER ' || V_CUSTNO || ' LAST MADE AN ORDER ON ' ||
                       TO_CHAR(V_ORDERDATE, 'DD/MM/YYYY') || '.');
	
  	-- CALCULATE THE DURATION WITH THE LAST ORDER MADE
  	V_DURATION := ROUND(SYSDATE - V_ORDERDATE, 0);
  
  	IF (V_DURATION < 30) THEN
  		DBMS_OUTPUT.PUT_LINE('THAT WAS ' || V_DURATION || ' DAYS AGO.');
    
    ELSE
  		V_MONTHS := MONTHS_BETWEEN(SYSDATE, V_ORDERDATE);
    	DBMS_OUTPUT.PUT_LINE('THAT WAS ' || V_MONTHS || ' MONTHS AGO.');
  	END IF;
  
  ELSE
  		-- CALL THE EXCEPTION SECTION
   	  -- NO_DATA_FOUND IS A DEFAULT 
  		RAISE NO_DATA_FOUND;    
  END IF;

  	-- EXCEPTION SECTION -- WITH DEFAULT EXCEPTION HANDLING
  EXCEPTION 
  	WHEN NO_DATA_FOUND THEN
			DBMS_OUTPUT.PUT_LINE('CUSTOMER NOT FOUND!');  
END;
/

-- HEADING SECTION WITH INPUT PARAMETER - ACCEPT ORDERNO AND PRODUCTCODE
CREATE OR REPLACE PROCEDURE PRC_DISCOUNT_CLAIM (V_ORDERNO IN NUMBER, V_PRODUCTCODE IN VARCHAR) IS

-- DECLARATION SECTION
V_ORDERDATE   ORDERS.ORDERDATE%TYPE;
V_ACTUALPRICE	ORDERDETAILS.PRICEEACH%TYPE;
V_BUYPRICE		PRODUCTS.BUYPRICE%TYPE;

V_DURATION  NUMBER(4);

-- EXECUTION SECTION -- WITH SELECT INTO STATEMENT AND NESTED IF ELSE STATEMENT

BEGIN
-- DISPLAY ORDERS INFORMATION
  -- EXTRACT CLAIM DATA -- CLAIM MUST BE LESS THAN 30 DAYS
  -- USE SELECT INTO STATEMENT AND FILTER BY INPUT PARAMETER
	SELECT ORDERDATE INTO V_ORDERDATE
  FROM   ORDERS
  WHERE  ORDERNUMBER = V_ORDERNO AND
  			 (SYSDATE - ORDERDATE) < 30;
  
  -- V_DURATION ONLY ACCEPT ORDERDATE LESS THAN 30 DAYS
  V_DURATION := ROUND(SYSDATE - V_ORDERDATE);
  
  -- DISPLAY ORDER INFORMATION TO THE CUSTOMER
  DBMS_OUTPUT.PUT_LINE('CUSTOMER LAST MADE AN ORDER ON ' || 
                       TO_CHAR(V_ORDERDATE, 'DD-MON-YYYY') || '.');
  DBMS_OUTPUT.PUT_LINE('THAT WAS ' || V_DURATION || ' DAYS AGO.');

	-- EXTRACT ORDERDETAILS AND PRODUCTS DATA 
	-- BUYPRICE = PRICE PURCHASED BY THE COMPANY
  -- PRICEEACH - PRICE THAT SOLD TO THE CUSTOMER
  
  -- EXTRACT ACTUAL PRICE (SELLING PRICE)
  SELECT PRICEEACH INTO V_ACTUALPRICE
  FROM   ORDERDETAILS
  WHERE  ORDERNUMBER = V_ORDERNO AND PRODUCTCODE = V_PRODUCTCODE;

	-- EXTRACT PURCHASE PRICE (BUYING PRICE)
  SELECT BUYPRICE INTO V_BUYPRICE 
  FROM   PRODUCTS
  WHERE  PRODUCTCODE = V_PRODUCTCODE;
  
  -- MAKE SURE AT LEAST EARN 5% 
  V_MINPRICE        := V_BUYPRICE * 1.05;
  V_DISCOUNTEDPRICE := V_ACTUALPRICE * 0.95;

-- NESTED IF ELSE STATEMENT -- CHECK THE ORDERDATE VALIDITY AND DISCOUNT VALIDITY
-- SELECT INTO STATEMENT -- DEAL WITH ORDERS, ORDERDETAILS AND PRODUCTS TABLE TO EXTRACT THE DATA
	 IF (V_DURATION < 30) THEN
   	IF (V_DISCOUNTEDPRICE >= V_MINPRICE) THEN
    	UPDATE ORDERDETAILS 
      SET    PRICEEACH = ROUND(V_DISCOUNTEDPRICE, 2)
      WHERE  ORDERNUMBER = V_ORDERNO AND
             PRODUCTCODE = V_PRODUCTCODE;
    	DBMS_OUTPUT.PUT_LINE('PAID : ' || V_ACTUALPRICE);
    	DBMS_OUTPUT.PUT_LINE('NEW DISCOUNTED PRICE : ' || V_DISCOUNTEDPRICE);
    
    ELSE
    	DBMS_OUTPUT.PUT_LINE('PAID : ' || V_ACTUALPRICE);
      DBMS_OUTPUT.PUT_LINE('ALREADY LOWEST PRICE!');
    END IF;
  ELSE
  	RAISE NO_DATA_FOUND;
  END IF;
  
-- EXCEPTION SECTION WITH CUSTOM EXCEPTION
	EXCEPTION
  	DBMS_OUTPUT.PUT_LINE('UNABLE TO PROCESS DISCOUNT');
    DBMS_OUTPUT.PUT_LINE('ORDER NOT FOUND OR MORE THAN 30 DAYS');
END;
/

CREATE OR REPLACE PROCEDURE prc_order_details AS
   -- VARIABLE DECLARATION
   v_orderNo		ORDERS.orderNumber%TYPE;
   v_orderDate		ORDERS.orderDate%TYPE;
   v_requiredDate	ORDERS.requiredDate%TYPE;
   v_shippedDate	ORDERS.shippedDate%TYPE;
   v_custNo		ORDERS.customerNumber%TYPE;
   v_productCode	ORDERDETAILS.productCode%TYPE;
   v_qtyOrd		ORDERDETAILS.quantityOrdered%TYPE;
   v_priceEach		ORDERDETAILS.priceEach%TYPE;

-- FIRST CURSOR DECLARATION
cursor order_cursor is
       select customerNumber, orderNumber, orderDate, requiredDate, shippedDate
       from ORDERS;
       
-- SECOND CURSOR DECLARATION
cursor orderDetail_cursor is
       select productCode, quantityOrdered, priceEach
       from ORDERDETAILS
       where orderNumber = v_orderNo;
-- EXECUTION SECTION        
BEGIN
   OPEN order_cursor;
   FETCH order_cursor
         INTO v_custNo, v_orderNo, v_orderDate, v_requiredDate, v_shippedDate;
   WHILE order_cursor%FOUND
   LOOP
   -- USE WHILE LOOP TO DISPLAY ALL OF THE CUSTOMERS WITH ORDERS MADE
      DBMS_OUTPUT.PUT_LINE('Customer No : '||v_custNo);
      DBMS_OUTPUT.PUT_LINE('Order    No : '||v_orderNo);
      DBMS_OUTPUT.PUT_LINE('Order Date     : '||v_orderDate);
      DBMS_OUTPUT.PUT_LINE('Shipped        : '||v_shippedDate);
      DBMS_OUTPUT.PUT_LINE('Required Date  : '||v_requiredDate);
      dbms_output.put_line(chr(10));

			
      OPEN orderDetail_cursor;
      FETCH orderDetail_cursor
            INTO v_productCode, v_qtyOrd, v_priceEach;

      WHILE orderDetail_cursor%FOUND 
      LOOP
      -- USE WHILE LOOP TO DISPLAY ALL OF THE ORDER DETAILS INFORMATION
         DBMS_OUTPUT.PUT_LINE(v_productCode||'***'||v_qtyOrd||'***'||v_priceEach);

         FETCH orderDetail_cursor
            INTO v_productCode, v_qtyOrd, v_priceEach;
      END LOOP;
      CLOSE orderDetail_cursor;

      DBMS_OUTPUT.PUT_LINE('End of Customer '||v_custNo||'************************');
      dbms_output.put_line(chr(10));

   FETCH order_cursor
         INTO v_custNo, v_orderNo, v_orderDate, v_requiredDate, v_shippedDate;
   END LOOP;
   CLOSE order_cursor;
END;

V_PRODUCTNAME		PRODUCTS.PRODUCTNAME%TYPE;

V_QTY						ORDERDETAILS.QUANTITYORDERED%TYPE;
V_PRICE					ORDERDETAILS.PRICEEACH%TYPE;

V_SUBTOTAL			NUMBER(9,2) := 0.0;
V_ORDERTOTAL		NUMBER(9,2) := 0.0;
V_GRANDTOTAL		NUMBER(11,2) := 0.0;

-- DECLARE A REF CURSOR
TYPE ORDER_CURSOR_TYP IS REF CURSOR;
ORDER_CURSOR ORDER_CURSOR_TYP; -- ORDER_CURSOR NOW IS A NESTED CURSOR VARIABLE


-- CURSOR DECLARATION SECTION
-- USE REF CURSOR -- NESTED CURSOR TO EXTRACT ORDERS AND ORDERDETAILS INFORMATION
-- OUTER CURSOR -- SELECT ORDERNUMBER AND ORDERDATE
-- INNER CURSOR -- SELECT PRODUCTCODE, PRODUCTNAME, QUANTITYORDERED, PRICEEACH
CURSOR C1  IS
	SELECT ORDERNUMBER, ORDERDATE, 
  			 CURSOR (SELECT P.PRODUCTCODE, PRODUCTNAME, QUANTITYORDERED, PRICEEACH
                 FROM   ORDERDETAILS OD, PRODUCTS P
                 WHERE  OD.PRODUCTCODE = P.PRODUCTCODE AND
                        OD.ORDERNUMBER = O.ORDERNUMBER)
  FROM ORDERS O
  WHERE ORDERDATE BETWEEN V_STARTDATE AND V_ENDDDATE;

-- EXECUTION SECTION
BEGIN
	-- OPEN THE CURSOR 
  OPEN C1;
  
  -- USE LOOP STATEMENT TO FETCH ALL THE VALUE IN C1
  LOOP
  	FETCH C1 INTO V_ORDERNUMBER, V_ORDERDATE, ORDER_CURSOR;
    EXIT WHEN C1%NOTFOUND;
    
    DBMS_OUTPUT.PUT_LINE(CHR(10));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 85, '='));
    DBMS_OUTPUT.PUT_LINE('ORDER NUMBER : ' || V_ORDERNUMBER);
    DBMS_OUTPUT.PUT_LINE('ORDER DATE   : ' || TO_CHAR(V_ORDERDATE, 'DD-MON-YYYY'));
    
    DBMS_OUTPUT.PUT_LINE(RPAD('.', 85, '.'));
    DBMS_OUTPUT.PUT_LINE(RPAD('PRODUCT CODE', 12, ' ') || ' ' || 
                         RPAD('PRODUCT NAME', 40, ' ') || ' ' || 
                         LPAD('QUANTITY', 10, ' ')     || ' ' ||
                         LPAD('PRICE EACH', 10, ' ')  || ' ' ||
                         LPAD('SUB TOTAL', 10, ' '));
    DBMS_OUTPUT.PUT_LINE(RPAD('.', 85, '.'));
    
		-- USE LOOP STATEMENT TO FETCH ALL THE VALUE IN ORDER_CURSOR
    LOOP
    	FETCH ORDER_CURSOR INTO V_PRODUCTCODE, V_PRODUCTNAME, V_QTY, V_PRICE;
      	EXIT WHEN ORDER_CURSOR%NOTFOUND;
        
        -- CALCULATE THE SUBTOTAL WITH QUANTITY * PRICEEACH
        V_SUBTOTAL   := V_QTY * V_PRICE;
        
        -- ACCUMULATE THE ORDERTOTAL IN LOOP
        V_ORDERTOTAL := V_ORDERTOTAL + V_SUBTOTAL;
        
   	 		DBMS_OUTPUT.PUT_LINE(RPAD(V_PRODUCTCODE, 12, ' ') || ' ' || 
                         		 RPAD(V_PRODUCTNAME, 40, ' ') || ' ' || 
                         	   LPAD(V_QTY, 10, ' ')     || ' ' ||
                         		 LPAD(TO_CHAR(V_PRICE, '$9,999.99'), 10, ' ')  || ' ' ||
                         		 LPAD(TO_CHAR(V_SUBTOTAL, '$9,999.99'), 10, ' '));        
        
        -- END OF INNER LOOP FROM ORDER_CURSOR
        END LOOP;
        
        -- PRINT OUT THE ACCUMULATED ORDERTOTAL FROM THE FIRST ORDER
        DBMS_OUTPUT.PUT_LINE(RPAD('*', 85, '*'));
        DBMS_OUTPUT.PUT_LINE('TOTAL PURCHASED ' || ORDER_CURSOR%ROWCOUNT || ' PRODUCTS');
        DBMS_OUTPUT.PUT_LINE('ORDER TOTAL : ' || TO_CHAR(V_ORDERTOTAL, '$999,999.99'));
        DBMS_OUTPUT.PUT_LINE(RPAD('*', 85, '*'));
        DBMS_OUTPUT.PUT_LINE(CHR(10));
        
        V_GRANDTOTAL := V_GRANDTOTAL + V_ORDERTOTAL;
      
      -- END OF OUTER LOOP FROM C1
      END LOOP;
    	DBMS_OUTPUT.PUT_LINE('GRAND TOTAL : ' || TO_CHAR(V_GRANDTOTAL, '$999,999,999.99'));
      
		-- CLOSE THE CURSOR
    CLOSE C1;
END;

CREATE OR REPLACE PROCEDURE SHOW_CUSTOMER IS

-- VARIABLE DECLARATION SECTION
V_ORDERTOTAL NUMBER(18,2) := 0.0;

-- CURSOR DECLARATION SECTION
-- FIRST CURSOR -- SHOW CUSTOMER AND ORDER INFORMATION (LAST ORDER MADE)
CURSOR LAST_ORDER IS
	SELECT O.CUSTOMERNUMBER, CUSTOMERNAME, O.ORDERNUMBER, ORDERDATE
  FROM   CUSTOMERS C, ORDERS O
  WHERE  C.CUSTOMERNUMBER = O.CUSTOMERNUMBER AND
  			 ORDERDATE = (SELECT MAX(ORDERDATE)
                      FROM   ORDERS O2
                      WHERE  O2.CUSTOMERNUMBER = O.CUSTOMERNUMBER AND
                     				 MONTHS_BETWEEN(SYSDATE, ORDERDATE) > 6)
  ORDER BY O.CUSTOMERNUMBER;

-- SECOND CURSOR -- SHOW LAST ORDER VALUE -- ORDERDETAILS TABLE
-- GET THE FILTERED VALUE FROM FIRST CURSOR
CURSOR ORDERDETAIL_CURSOR(V_ORDERNO IN NUMBER) IS
	SELECT QUANTITYORDERED, PRICEEACH, QUANTITYORDERED * PRICEEACH AS SUBTOTAL
  FROM   ORDERDETAILS
  WHERE  ORDERNUMBER = V_ORDERNO;

-- FORM AN ARRAY STRUCTURE AND COPY THE STRUCTURE FROM LAST_ORDER CURSOR
LASTORDERREC		LAST_ORDER%ROWTYPE;
ORDERDETAILREC	ORDERDETAIL_CURSOR%ROWTYPE;

-- EXECUTION SECTION
BEGIN
	-- USE CURSOR FOR LOOP TO EXTRACT ALL THE VALUES
  FOR LASTORDERREC IN LAST_ORDER LOOP
  	DBMS_OUTPUT.PUT_LINE(CHR(10));
  	DBMS_OUTPUT.PUT_LINE(RPAD('=', 100, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('CUST NO', 12, ' ')    || ' ' ||
                         RPAD('CUST NAME', 30, ' ')  || ' ' ||
                         RPAD('ORDER NO', 12, ' ')   || ' ' ||
                         RPAD('ORDER DATE', 15, ' '));    
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 100, '='));
  
		-- DISPLAY THE CUSTOMER ORDER INFORMATION
  	DBMS_OUTPUT.PUT_LINE(RPAD(LASTORDERREC.CUSTOMERNUMBER, 12, ' ')  || ' ' ||
                         RPAD(LASTORDERREC.CUSTOMERNAME, 30, ' ')    || ' ' ||
                         RPAD(LASTORDERREC.ORDERNUMBER, 12, ' ')     || ' ' ||
                         RPAD(TO_CHAR(LASTORDERREC.ORDERDATE, 'DD-MON-YYYY'), 15, ' '));  

		DBMS_OUTPUT.PUT_LINE(RPAD('*', 100, '*'));
    
    -- DISPLAY THE ORDER DETAILS VALUE
		V_ORDERTOTAL := 0.0;
    
    DBMS_OUTPUT.PUT_LINE(RPAD('QUANTITY', 12)  || ' ' ||
                         RPAD('PRICE', 15) 		 || ' ' ||
                         RPAD('SUB TOTAL', 15));
		DBMS_OUTPUT.PUT_LINE(RPAD('.', 100, '.'));

		-- USE ANOTHER FOR LOOP TO DISPLAY ORDER DETAILS VALUE
    FOR ORDERDETAILREC IN ORDERDETAIL_CURSOR(LASTORDERREC.ORDERNUMBER) LOOP
    	DBMS_OUTPUT.PUT_LINE(RPAD(ORDERDETAILREC.QUANTITYORDERED, 12)  || ' ' ||
                           RPAD(TO_CHAR(ORDERDETAILREC.PRICEEACH, '$999,999.99'), 15) 		 || ' ' ||
                        	 RPAD(TO_CHAR(ORDERDETAILREC.SUBTOTAL, '%999,999,999.99'), 15));
      
      V_ORDERTOTAL := V_ORDERTOTAL + ORDERDETAILREC.SUBTOTAL;
		END LOOP;
    
    DBMS_OUTPUT.PUT_LINE(CHR(10));
    DBMS_OUTPUT.PUT_LINE('ORDER TOTAL OF THE LAST ORDER IS ' || TO_CHAR(V_ORDERTOTAL, '$999,999,999.99'));
		
    DBMS_OUTPUT.PUT_LINE(RPAD('.', 100, '.'));
  END LOOP;
END;

-- STEP 2 -- CREATE THE FIRE ONCE TRIGGER
CREATE OR REPLACE TRIGGER Track_Employees
BEFORE INSERT OR
       UPDATE OR
       DELETE
       ON employees
DECLARE
   v_date date; 

BEGIN
   select TransDate into v_date
   from Employees_Audit
   where to_char(TransDate,'DD-MON-YY') = to_char(SYSDATE,'DD-MON-YY');
-- check if there is already a record for today ---
   
   if SQL%FOUND THEN
      CASE
         WHEN INSERTING THEN
            UPDATE Employees_AUDIT
               SET No_Of_Insert = No_Of_Insert + 1
            where to_char(TransDate,'DD-MON-YY') = to_char(SYSDATE,'DD-MON-YY');
         WHEN UPDATING THEN
            UPDATE Employees_AUDIT
               SET No_Of_Update = No_Of_Update + 1
            where to_char(TransDate,'DD-MON-YY') = to_char(SYSDATE,'DD-MON-YY');
         WHEN DELETING THEN
            UPDATE Employees_AUDIT
               SET No_Of_Delete = No_Of_Delete + 1
            where to_char(TransDate,'DD-MON-YY') = to_char(SYSDATE,'DD-MON-YY');
      END CASE;
   end if;

   exception
      when no_data_found then
      CASE
         WHEN INSERTING THEN
            INSERT INTO Employees_AUDIT VALUES(sysdate,1,0,0);
         WHEN UPDATING THEN
            INSERT INTO Employees_AUDIT VALUES(sysdate,0,1,0);
         WHEN DELETING THEN
            INSERT INTO Employees_AUDIT VALUES(sysdate,0,0,1);
      END CASE;
END;

 -- CREATE A BEFORE TRIGGER TO VALIDATE THE EMPLOYEES INFORMATION
CREATE OR REPLACE TRIGGER TRG_CHK_EMP
BEFORE INSERT ON EMPLOYEES
FOR EACH ROW 

DECLARE 
	V_DATEDIFF	NUMBER;
  V_AGE NUMBER;

BEGIN
	-- PASS THE DATA OUT TO THE EXTERNAL ENVIRONMENT 
	:NEW.HIREDATE := SYSDATE;
  
  -- DATE DIFFERENCE IS IN MONTH (MONTH VALUE CALCULATION)
  -- COMPARE CURRENT DATE WITH DATEOFBIRTH FROM DATABASE
  -- CALCULATE DATE DIFFERENCE IN MONTH VALUE
  V_DATEDIFF := MONTHS_BETWEEN(SYSDATE, :NEW.DATEOFBIRTH);
  
  -- CALCULATE AGE WITH DATEDIFF DIVIDED BY 12 INTO YEAR VALUE
  V_AGE			 := V_DATEDIFF / 12;
  
  DBMS_OUTPUT.PUT_LINE('NEW EMPLOYEE IS ' || ROUND(V_AGE, 2) || ' YEARS OLD.');
	
  -- USE IF ELSE STATEMENT TO COMPARE THE AGE 
  IF V_AGE < 18 THEN
  	RAISE_APPLICATION_ERROR(-20000, 'CANNOT INSERT EMPLOYEE BELOW 18 YEARS OLD!');
  END IF;
END;
/

	ALTER TABLE EMPLOYEES
  ADD CONSTRAINT SAL_GRADE_FK FOREIGN KEY (SAL_CODE) REFERENCES SALARY_GRADE(GRADE_CODE);


SQL> DESC SALARY_GRADE;

CREATE OR REPLACE TRIGGER TRG_CHK_SALARY
BEFORE INSERT OR UPDATE OF SALARY ON EMPLOYEES

FOR EACH ROW

DECLARE
	V_MIN	SALARY_GRADE.MIN_SALARY%TYPE;
  V_MAX	SALARY_GRADE.MAX_SALARY%TYPE;

BEGIN
	-- EXTRACT MINIMUM AND MAXIMUM SALARY FROM SALARY_GRAGE TABLE
  -- FOR RANGE COMPARISON WITH SELECT INTO STATEMENT 
  SELECT MIN_SALARY, MAX_SALARY INTO V_MIN, V_MAX
	FROM   SALARY_GRADE
  WHERE  GRADE_CODE = :NEW.SAL_CODE;
	-- SAL_CODE FROM EMPLOYEES TABLE WILL PASSED IN THE VALUE INTO
  -- EXECUTION FOR COMPARISON IN EXTRACTING MIN AND MAX SALARY
  -- FROM SALARY_GRADE TABLE
  
  -- USE IF NOT STATEMENT TO VALIDATE THE SALARY RANGE
  -- SALARY FROM EMPLOYEES TABLE WILL COMPARE WITH SALARY RANGE FROM SALARY_GRADE TABLE
  IF NOT (:NEW.SALARY BETWEEN V_MIN AND V_MAX) THEN
		RAISE_APPLICATION_ERROR(-20000, 'SALARY NOT IN THE CORRECT RANGE');
	END IF;
END;
/
