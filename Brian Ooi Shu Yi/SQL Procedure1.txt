CREATE OR REPLACE PROCEDURE calculate_dynamic_delivery_fee (
    p_order_id IN VARCHAR2,
    p_use_voucher IN CHAR DEFAULT 'Y'
) AS
    v_member_id       ORDERS.MemberID%TYPE;
    v_delivery_id     ORDERS.DeliveryID%TYPE;
    v_mem_voucher_id  ORDERS.MemberVoucherID%TYPE;
    v_order_total     ORDERS.TotalAmount%TYPE;
    v_distance        DELIVERY.DistanceInKm%TYPE;
    v_speed_fee       DELIVERYTYPE.DeliveryFee%TYPE;
    v_delivery_name   DELIVERYTYPE.DeliveryName%TYPE;
    
    -- STAGE 1:
    v_base_dist_fee     NUMBER(6,2) := 4.99; 
    v_dist_surcharge    NUMBER(6,2) := 0;
    v_distance_subtotal NUMBER(6,2) := 0; 
    v_original_fee      NUMBER(6,2) := 0;
    
    -- STAGE 2:
    v_final_fee       NUMBER(6,2) := 0;  
    v_vip_count       NUMBER := 0;
    v_vip_discount    NUMBER(6,2) := 0;
    v_vip_pct         NUMBER(6,2) := 0;
    

    v_voucher_type    VOUCHER.VoucherTypeID%TYPE;
    v_voucher_is_used MEMBERVOUCHER.IsUsed%TYPE;
    v_flat_discount   VOUCHER.DiscountAmount%TYPE := 0;
    v_voucher_waived  NUMBER(6,2) := 0;
    v_voucher_msg     VARCHAR2(100) := 'No Voucher Attached';
    
    e_no_delivery     EXCEPTION;
BEGIN
    SELECT O.MemberID, O.DeliveryID, O.MemberVoucherID, O.TotalAmount, D.DistanceInKm, DT.DeliveryFee, DT.DeliveryName
    INTO v_member_id, v_delivery_id, v_mem_voucher_id, v_order_total, v_distance, v_speed_fee, v_delivery_name
    FROM ORDERS O
    LEFT JOIN DELIVERY D ON O.DeliveryID = D.DeliveryID
    LEFT JOIN DELIVERYTYPE DT ON D.DeliveryTypeID = DT.DeliveryTypeID
    WHERE O.OrderID = p_order_id;

    IF v_delivery_id IS NULL THEN
        RAISE e_no_delivery;
    END IF;

    -- 1) & 2) STAGE 1: ORIGINAL DELIVERY FEE
    IF v_distance > 5 THEN
        v_dist_surcharge := CEIL(v_distance - 5) * 1.50;
    END IF;

    v_distance_subtotal := v_base_dist_fee + v_dist_surcharge;
    v_original_fee := v_distance_subtotal + v_speed_fee;
    v_final_fee := v_original_fee;

    -- 3) & 4) STAGE 2: DISCOUNTS
    SELECT COUNT(*) INTO v_vip_count
    FROM SUBSCRIPTION
    WHERE MemberID = v_member_id AND Status = 'Active';

    IF v_vip_count > 0 THEN
        v_vip_pct := FLOOR(NVL(v_order_total, 0) / 10) * 0.05;
        IF v_vip_pct > 0.50 THEN v_vip_pct := 0.50; END IF;

        v_vip_discount := ROUND(v_final_fee * v_vip_pct, 2);
        v_final_fee := v_final_fee - v_vip_discount;
    END IF;

    -- Voucher Processing
    IF v_mem_voucher_id IS NOT NULL THEN
        SELECT V.VoucherTypeID, MV.IsUsed, NVL(V.DiscountAmount, 0)
        INTO v_voucher_type, v_voucher_is_used, v_flat_discount
        FROM MEMBERVOUCHER MV
        JOIN VOUCHER V ON MV.VoucherID = V.VoucherID
        WHERE MV.MemberVoucherID = v_mem_voucher_id;

        v_voucher_msg := 'Detected: ' || v_mem_voucher_id;

        IF p_use_voucher = 'Y' THEN
            IF v_voucher_is_used = 'N' THEN
                IF v_voucher_type = 'VT00003' THEN
                    v_voucher_waived := v_flat_discount;
                    IF v_voucher_waived > v_final_fee THEN
                        v_voucher_waived := v_final_fee;
                    END IF;
                    
                    v_final_fee := v_final_fee - v_voucher_waived; 
                    v_voucher_msg := v_voucher_msg || ' -> USER APPLIED! (-RM ' || TO_CHAR(v_voucher_waived, 'FM99,990.00') || ')';
                    UPDATE MEMBERVOUCHER SET IsUsed = 'Y' WHERE MemberVoucherID = v_mem_voucher_id;
                ELSE
                    v_voucher_msg := v_voucher_msg || ' -> NOT APPLICABLE (Not a Delivery Voucher)';
                END IF;
            ELSE
                v_voucher_msg := v_voucher_msg || ' -> REJECTED (Voucher Already Used!)';
            END IF;
        ELSE
            v_voucher_msg := v_voucher_msg || ' -> USER DECLINED (Saved for next time)';
        END IF;
    END IF;

    IF v_final_fee < 0 THEN v_final_fee := 0; END IF;

    UPDATE DELIVERY 
    SET OriginalDeliveryPrice = v_original_fee,
        DiscountedDeliveryPrice = v_final_fee 
    WHERE DeliveryID = v_delivery_id;
    COMMIT;

    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    DBMS_OUTPUT.PUT_LINE('                 DYNAMIC DELIVERY FEE ALGORITHM                 ');
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    
    DBMS_OUTPUT.PUT_LINE(RPAD('Order ID', 44) || ': ' || UPPER(p_order_id));
    DBMS_OUTPUT.PUT_LINE(RPAD('Food Total Amount', 44) || ': RM ' || LPAD(TO_CHAR(v_order_total, 'FM99,990.00'), 9));
    DBMS_OUTPUT.PUT_LINE(RPAD('Total Distance (KM)', 44) || ':    ' || LPAD(TO_CHAR(v_distance, 'FM990.00'), 9));
    
    DBMS_OUTPUT.PUT_LINE(RPAD('-', 64, '-'));
    DBMS_OUTPUT.PUT_LINE('                  [ STAGE 1: ORIGINAL FEE ]                     ');
    
    DBMS_OUTPUT.PUT_LINE(RPAD('1) Base Fee (First 5km)', 44) || ': RM ' || LPAD(TO_CHAR(v_base_dist_fee, 'FM99,990.00'), 9));
    
    IF v_dist_surcharge > 0 THEN
        DBMS_OUTPUT.PUT_LINE(RPAD('   Distance Surcharge (RM 1.50/km > 5km)', 44) || ': RM ' || LPAD(TO_CHAR(v_dist_surcharge, 'FM99,990.00'), 9));
    END IF;
    
    DBMS_OUTPUT.PUT_LINE(RPAD('.', 64, '.'));
    DBMS_OUTPUT.PUT_LINE(RPAD('>> DISTANCE SUBTOTAL', 44) || ': RM ' || LPAD(TO_CHAR(v_distance_subtotal, 'FM99,990.00'), 9));
    DBMS_OUTPUT.PUT_LINE(RPAD('.', 64, '.'));
    
    DBMS_OUTPUT.PUT_LINE(RPAD('2) Delivery Speed (' || v_delivery_name || ')', 44) || ': RM ' || LPAD(TO_CHAR(v_speed_fee, 'FM99,990.00'), 9));
    
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('>> ORIGINAL DELIVERY FEE', 44) || ': RM ' || LPAD(TO_CHAR(v_original_fee, 'FM99,990.00'), 9));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    
    DBMS_OUTPUT.PUT_LINE('                     [ STAGE 2: DISCOUNTS ]                     ');
    
    IF v_vip_count > 0 THEN
        DBMS_OUTPUT.PUT_LINE(RPAD('3) VIP Tiered Discount (' || (v_vip_pct * 100) || '%)', 44) || ':-RM ' || LPAD(TO_CHAR(v_vip_discount, 'FM99,990.00'), 9));
    ELSE
        DBMS_OUTPUT.PUT_LINE(RPAD('3) VIP Discount', 44) || ': Not Applicable');
    END IF;
    
    DBMS_OUTPUT.PUT_LINE('4) Vouchers:');
    DBMS_OUTPUT.PUT_LINE('   ' || v_voucher_msg);
    
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    DBMS_OUTPUT.PUT_LINE(RPAD('>> FINAL DELIVERY FEE', 44) || ': RM ' || LPAD(TO_CHAR(v_final_fee, 'FM99,990.00'), 9));
    DBMS_OUTPUT.PUT_LINE(RPAD('=', 64, '='));
    DBMS_OUTPUT.PUT_LINE('>> DB Status: DELIVERY table updated successfully.');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('ERROR: Order ID ' || p_order_id || ' does not exist or is missing required delivery details.');
    WHEN e_no_delivery THEN
        DBMS_OUTPUT.PUT_LINE('ERROR: Order ' || p_order_id || ' is a Self-Pickup order. No delivery fee needed.');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An unexpected error occurred: ' || SQLERRM);
END;
