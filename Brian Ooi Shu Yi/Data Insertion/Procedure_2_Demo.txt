PROMPT =================================================================
PROMPT [ BEFORE EXECUTION ] Checking Current VIP Status & Wallet
PROMPT =================================================================
PROMPT >> 1. Current VIP Status for Member:

SELECT M.MemberID, M.FullName, NVL(SP.PlanName, 'Not a VIP') AS PLANNAME, NVL(S.Status, 'None') AS STATUS
FROM MEMBER M
LEFT JOIN SUBSCRIPTION S ON M.MemberID = S.MemberID AND S.Status = 'Active'
LEFT JOIN SUBSCRIPTIONPLAN SP ON S.SubscriptionPlanID = SP.SubscriptionPlanID
WHERE M.MemberID = 'M00295';

PROMPT
PROMPT >> 2. Current Vouchers in Member's Wallet:

SELECT 
    NVL(MV.MemberVoucherID, 'EMPTY') AS WALLET_ID, 
    NVL(V.VoucherName, 'No Vouchers') AS VOUCHER_NAME,
    NVL(V.VoucherCode, '-') AS VOUCHER_CODE, 
    NVL(TO_CHAR(MV.DateClaimed, 'DD-MON-YYYY'), '-') AS DATE_CLAIMED, 
    NVL(MV.IsUsed, '-') AS USED
FROM MEMBER M
LEFT JOIN MEMBERVOUCHER MV ON M.MemberID = MV.MemberID
LEFT JOIN VOUCHER V ON MV.VoucherID = V.VoucherID
LEFT JOIN VOUCHERTYPE VT ON V.VoucherTypeID = VT.VoucherTypeID
WHERE M.MemberID = 'M00295'
ORDER BY MV.MemberVoucherID;
