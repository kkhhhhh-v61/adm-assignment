-- Query 2: Lost Revenue from Incomplete Orders based on Restaurant (Summary of the total potential revenue lost from orders that are "Cancelled" status)

-- ENABLE FORMATTING
CL SCR;
SET SQLBLANKLINES ON;
SET SERVEROUTPUT ON;

-- Create View for Cancelled Orders grouped by Restaurant
-- We use a CASE statement to handle the discount logic for Sets vs Individual Items
CREATE OR REPLACE VIEW vw_LostRevenueByRest AS
SELECT  R.RestaurantName AS Restaurant_Name,
        SUM(OI.Quantity) AS Total_Items_Lost,
        SUM(
            CASE 
                WHEN OI.ItemSetID IS NULL THEN (OI.Quantity * OI.PriceEach)
                ELSE (OI.Quantity * OI.PriceEach - NVL(S.Discount, 0))
            END
        ) AS Lost_Revenue
FROM    ORDERS O
JOIN    ORDERITEM OI ON O.OrderID = OI.OrderID
JOIN    ITEM I ON OI.ItemID = I.ItemID
JOIN    RESTAURANT R ON I.RestaurantID = R.RestaurantID
LEFT JOIN ITEMSET S ON OI.ItemSetID = S.ItemSetID
WHERE   O.OrderStatus = 'Cancelled'
GROUP BY R.RestaurantName;

CL SCR;

-- DISPLAY QUERY RESULTS

SET LINESIZE 100
SET FEEDBACK OFF

COLUMN RESTAURANT_NAME  FORMAT A33 HEADING 'RESTAURANT NAME'    JUSTIFY LEFT
COLUMN ITEMS_LOST       FORMAT A16 HEADING 'TOTAL QTY LOST'     JUSTIFY RIGHT
COLUMN REVENUE_LOST     FORMAT A18 HEADING 'LOST REVENUE (RM)'  JUSTIFY RIGHT
COLUMN IMPACT_PCT       FORMAT A12 HEADING 'IMPACT (%)'         JUSTIFY RIGHT

TTITLE LEFT '==================================================================================' SKIP 1 -
       LEFT '                  LOST REVENUE FROM CANCELLED ORDERS BY RESTAURANT                ' SKIP 1 -
       LEFT '==================================================================================' SKIP 1

-- Using UNION ALL and DUAL to create the formatted summary table
SELECT RESTAURANT_NAME, ITEMS_LOST, REVENUE_LOST, IMPACT_PCT
FROM (
    SELECT  Restaurant_Name AS RESTAURANT_NAME,
            LPAD(TRIM(TO_CHAR(Total_Items_Lost, '999,990')), 16) AS ITEMS_LOST,
            LPAD(TRIM(TO_CHAR(Lost_Revenue, '$999,990.99')), 18) AS REVENUE_LOST,
            LPAD(TRIM(TO_CHAR(ROUND(RATIO_TO_REPORT(Lost_Revenue) OVER () * 100, 2), '990.00')), 12) AS IMPACT_PCT,
            1 AS SortOrder,
            Lost_Revenue AS SortRev
    FROM    vw_LostRevenueByRest

    UNION ALL

    -- Visual Separator
    SELECT  '---------------------------------' AS RESTAURANT_NAME,
            '----------------' AS ITEMS_LOST,
            '------------------' AS REVENUE_LOST,
            '------------' AS IMPACT_PCT,
            2 AS SortOrder,
            0 AS SortRev
    FROM    DUAL

    UNION ALL

    -- Grand Total Row
    SELECT  'GRAND TOTAL LOST :' AS RESTAURANT_NAME,
            LPAD(TRIM(TO_CHAR(SUM(Total_Items_Lost), '999,990')), 16) AS ITEMS_LOST,
            LPAD(TRIM(TO_CHAR(SUM(Lost_Revenue), '$999,990.99')), 18) AS REVENUE_LOST,
            LPAD('100.00', 12) AS IMPACT_PCT,
            3 AS SortOrder,
            0 AS SortRev
    FROM    vw_LostRevenueByRest
)
ORDER BY SortOrder ASC, SortRev DESC;

TTITLE OFF
SET HEADING OFF

-- PL/SQL EXECUTIVE SUMMARY
DECLARE
  v_WorstRest     VARCHAR2(150);
  v_WorstRevenue  NUMBER(11,2);
  v_WorstPct      NUMBER(5,2);
BEGIN
  -- Extract the restaurant with the highest lost revenue
  SELECT Restaurant_Name, Lost_Revenue, ImpactPct
  INTO v_WorstRest, v_WorstRevenue, v_WorstPct
  FROM (
    SELECT Restaurant_Name,
           Lost_Revenue,
           ROUND(RATIO_TO_REPORT(Lost_Revenue) OVER () * 100, 2) AS ImpactPct
    FROM vw_LostRevenueByRest
    ORDER BY Lost_Revenue DESC
  )
  WHERE ROWNUM = 1;

  DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------');
  DBMS_OUTPUT.PUT_LINE('[ EXECUTIVE SUMMARY ]');
  DBMS_OUTPUT.PUT_LINE('>> MOST AFFECTED RESTAURANT : ' || INITCAP(v_WorstRest));
  DBMS_OUTPUT.PUT_LINE('>> TOTAL REVENUE LOST       : RM ' || TRIM(TO_CHAR(v_WorstRevenue, '999,990.99')) || ' (' || v_WorstPct || '% of Total Loss)');
  DBMS_OUTPUT.PUT_LINE('==================================================================================');

EXCEPTION
  -- Adjusted the exception to sound like a positive business outcome if no data is found
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('EXCELLENT NEWS: No cancelled orders found. Zero revenue lost.');
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('AN UNEXPECTED ERROR OCCURRED: ' || SQLERRM);
END;
/

SET HEADING ON
SET FEEDBACK ON
CLEAR COLUMNS
