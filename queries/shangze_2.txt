CL SCR

-- Create view to combine Package and Item Set popularity
CREATE OR REPLACE VIEW VIEW_PACKAGE_SET_POPULARITY AS
SELECT
    'PACKAGE' AS PRODUCT_TYPE,
    P.PackageName AS PRODUCT_NAME,
    SUM(OP.Quantity) AS TOTAL_ORDERED,
    SUM(OP.Quantity * P.DiscountedPrice) AS TOTAL_REVENUE
FROM PACKAGE P
JOIN ORDERPACKAGE OP ON P.PackageID = OP.PackageID
JOIN ORDERS O ON OP.OrderID = O.OrderID
WHERE O.OrderStatus = 'Completed'
GROUP BY P.PackageName

UNION ALL

SELECT
    'ITEM SET' AS PRODUCT_TYPE,
    S.SetName AS PRODUCT_NAME,
    SUM(OI.Quantity) AS TOTAL_ORDERED,
    SUM(OI.Quantity * OI.PriceEach) AS TOTAL_REVENUE
FROM ITEMSET S
JOIN ORDERITEM OI ON S.ItemSetID = OI.ItemSetID
JOIN ORDERS O ON OI.OrderID = O.OrderID
WHERE O.OrderStatus = 'Completed'
GROUP BY S.SetName;

-- SQL*Plus settings
SET LINESIZE 110
SET PAGESIZE 200
SET FEEDBACK OFF
SET SQLBLANKLINES ON
SET SERVEROUTPUT ON

-- Format columns
COLUMN PRODUCT_TYPE FORMAT A15 HEADING 'PRODUCT TYPE'
COLUMN PRODUCT_NAME FORMAT A30 HEADING 'PACKAGE / SET NAME'
COLUMN TOTAL_ORDERED FORMAT 999,990 HEADING 'TOTAL ORDERED'
COLUMN POPULARITY_SHARE FORMAT 990.00 HEADING 'POPULARITY (%)'
COLUMN TOTAL_REVENUE FORMAT $999,990.99 HEADING 'REVENUE (RM)'

-- Report header
TTITLE LEFT '========================================================================================' SKIP 1 -
       LEFT '                       PACKAGE AND SET POPULARITY ANALYSIS                              ' SKIP 1 -
       LEFT '========================================================================================' SKIP 2

-- Calculate totals
BREAK ON REPORT
COMPUTE SUM LABEL 'GRAND TOTAL :' OF TOTAL_ORDERED TOTAL_REVENUE ON REPORT

-- Main query
SELECT
    PRODUCT_TYPE,
    PRODUCT_NAME,
    TOTAL_ORDERED,
    ROUND(RATIO_TO_REPORT(TOTAL_ORDERED) OVER () * 100, 2) AS POPULARITY_SHARE,
    TOTAL_REVENUE
FROM VIEW_PACKAGE_SET_POPULARITY
ORDER BY TOTAL_ORDERED DESC;

-- Disable headers for summary
TTITLE OFF
CLEAR BREAKS
CLEAR COMPUTES
SET HEADING OFF

-- PL/SQL block for executive summary
DECLARE
    v_top_product VARCHAR2(100);
    v_top_type VARCHAR2(20);
    v_top_qty NUMBER;
    v_top_rev NUMBER;
BEGIN

    SELECT PRODUCT_NAME, PRODUCT_TYPE, TOTAL_ORDERED, TOTAL_REVENUE
    INTO v_top_product, v_top_type, v_top_qty, v_top_rev
    FROM (
        SELECT PRODUCT_NAME, PRODUCT_TYPE, TOTAL_ORDERED, TOTAL_REVENUE
        FROM VIEW_PACKAGE_SET_POPULARITY
        ORDER BY TOTAL_ORDERED DESC
    )
    WHERE ROWNUM = 1;

    DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(' [ PACKAGE / SET POPULARITY SUMMARY ]');
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE(' >> MOST POPULAR PRODUCT : ' || v_top_product);
    DBMS_OUTPUT.PUT_LINE(' >> PRODUCT TYPE         : ' || v_top_type);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL ORDERS         : ' || v_top_qty);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL REVENUE        : RM ' || v_top_rev);
    DBMS_OUTPUT.PUT_LINE('----------------------------------------------------------------------------------------');

END;
/

-- Reset SQL settings
SET HEADING ON
SET FEEDBACK ON
CLEAR COLUMNS
