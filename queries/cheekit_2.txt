CLEAR SCREEN;

SET LINESIZE 150;
SET PAGESIZE 50000;

TTITLE CENTER '========================================================================' SKIP 1 -
       CENTER 'MENU ITEM DEAD STOCK AND UNDERPERFORMANCE SUMMARY BY RESTAURANT' SKIP 1 -
       CENTER '========================================================================' SKIP 2

COLUMN "Restaurant" FORMAT A25;
COLUMN "Item ID" FORMAT A10;
COLUMN "Item Name" FORMAT A28;
COLUMN "Price(RM)" FORMAT 990.00;
COLUMN "Total Qty Sold" FORMAT 999,990;
COLUMN "Total Revenue(RM)" FORMAT 99,990.00;
COLUMN "Performance Status" FORMAT A32;

BREAK ON "Restaurant" SKIP 1;

SELECT 
    r.RestaurantName AS "Restaurant",
    i.ItemID AS "Item ID",
    i.ItemName AS "Item Name",
    i.Price AS "Price(RM)",
    NVL(SUM(oi.Quantity), 0) AS "Total Qty Sold",
    NVL(SUM(oi.Quantity * i.Price), 0) AS "Total Revenue(RM)",
    CASE 
        WHEN NVL(SUM(oi.Quantity), 0) = 0 THEN '🚨 DEAD STOCK (Never Ordered)'
        ELSE '⚠️ CRITICAL LOW (Under RM 20)'
    END AS "Performance Status"
FROM 
    ITEM i
JOIN 
    RESTAURANT r ON i.RestaurantID = r.RestaurantID
LEFT JOIN 
    ORDERITEM oi ON i.ItemID = oi.ItemID
LEFT JOIN 
    ORDERS o ON oi.OrderID = o.OrderID AND o.OrderStatus != 'Cancelled'
WHERE 
    i.IsAvailable = 'Y' 
GROUP BY 
    r.RestaurantName,
    i.ItemID,
    i.ItemName,
    i.Price
HAVING 
    NVL(SUM(oi.Quantity), 0) = 0 
    OR NVL(SUM(oi.Quantity * i.Price), 0) < 0.00
ORDER BY 
    "Restaurant" ASC,
    "Total Revenue(RM)" ASC, 
    "Total Qty Sold" ASC;

TTITLE OFF;
CLEAR BREAKS;
