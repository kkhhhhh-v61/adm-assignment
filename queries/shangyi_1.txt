CL SCR

-- Create view for category sales analysis
CREATE OR REPLACE VIEW VIEW_CATEGORY_SALES AS
SELECT
    C.CategoryName,
    COUNT(DISTINCT O.OrderID) AS TOTAL_ORDERS,
    SUM(OI.Quantity * OI.PriceEach) AS TOTAL_REVENUE
FROM CATEGORY C
JOIN ITEM I ON C.CategoryID = I.CategoryID
JOIN ORDERITEM OI ON I.ItemID = OI.ItemID
JOIN ORDERS O ON OI.OrderID = O.OrderID
WHERE O.OrderStatus = 'Completed'
GROUP BY C.CategoryName;

-- SQL*Plus settings
SET LINESIZE 110
SET PAGESIZE 100
SET FEEDBACK OFF
SET SQLBLANKLINES ON
SET SERVEROUTPUT ON

-- Format columns
COLUMN CATEGORYNAME FORMAT A30 HEADING 'FOOD CATEGORY'
COLUMN TOTAL_ORDERS FORMAT 999,990 HEADING 'TOTAL ORDERS'
COLUMN ORDER_SHARE FORMAT 990.00 HEADING 'ORDER SHARE (%)'
COLUMN TOTAL_REVENUE FORMAT $999,990.99 HEADING 'REVENUE (RM)'
COLUMN REVENUE_SHARE FORMAT 990.00 HEADING 'REVENUE SHARE (%)'

-- Report header
TTITLE LEFT '==========================================================================================' SKIP 1 -
       LEFT '                         CATEGORY SALES CONTRIBUTION ANALYSIS                             ' SKIP 1 -
       LEFT '==========================================================================================' SKIP 2

-- Calculate totals
BREAK ON REPORT
COMPUTE SUM LABEL 'GRAND TOTAL :' OF TOTAL_ORDERS TOTAL_REVENUE ON REPORT

-- Main query
SELECT
    CategoryName,
    TOTAL_ORDERS,
    ROUND(RATIO_TO_REPORT(TOTAL_ORDERS) OVER () * 100, 2) AS ORDER_SHARE,
    TOTAL_REVENUE,
    ROUND(RATIO_TO_REPORT(TOTAL_REVENUE) OVER () * 100, 2) AS REVENUE_SHARE
FROM VIEW_CATEGORY_SALES
ORDER BY TOTAL_REVENUE DESC;

-- Disable headers for summary
TTITLE OFF
CLEAR BREAKS
CLEAR COMPUTES
SET HEADING OFF

-- PL/SQL executive summary
DECLARE
    v_top_category VARCHAR2(100);
    v_top_rev NUMBER;
    v_top_pct NUMBER;
BEGIN

    SELECT CategoryName, TOTAL_REVENUE, REVENUE_SHARE
    INTO v_top_category, v_top_rev, v_top_pct
    FROM (
        SELECT 
            CategoryName,
            TOTAL_REVENUE,
            ROUND(RATIO_TO_REPORT(TOTAL_REVENUE) OVER () * 100, 2) AS REVENUE_SHARE
        FROM VIEW_CATEGORY_SALES
        ORDER BY TOTAL_REVENUE DESC
    )
    WHERE ROWNUM = 1;

    DBMS_OUTPUT.PUT_LINE('------------------------------------------------------------------------------------------');
    DBMS_OUTPUT.PUT_LINE(' [ CATEGORY SALES SUMMARY ]');
    DBMS_OUTPUT.PUT_LINE(' ');
    DBMS_OUTPUT.PUT_LINE(' >> TOP PERFORMING CATEGORY : ' || v_top_category);
    DBMS_OUTPUT.PUT_LINE(' >> TOTAL REVENUE GENERATED : RM ' || v_top_rev);
    DBMS_OUTPUT.PUT_LINE(' >> MARKET CONTRIBUTION     : ' || v_top_pct || '%');
    DBMS_OUTPUT.PUT_LINE('------------------------------------------------------------------------------------------');

END;
/

-- Reset SQL settings
SET HEADING ON
SET FEEDBACK ON
CLEAR COLUMNS
